/-! # Actual source type likelihood at Bob's decision

The source likelihood is computed from the genuine initialized behavioral
execution. Bob's information excludes Alice's immutable private type, while
retaining her public publication result. No native likelihood or equilibrium
is asserted.
-/

noncomputable section

namespace Vegas.PrivateResolutionFork

open SourceProgram GameTheory GameTheory.Protocol GameTheory.Math.Probability

def sourceBobInput (disclose : Bool) : setup.ProtocolView bob :=
  setup.protocolObserve bob (SourcePosition.guessed true disclose).state

def sourceBobType : setup.ProtocolState → Option Bool
  | some (.inr (.inr (.inr (.inl current)))) =>
      some (current.state.get (.there (.there (.there (.there (.there .here))))))
  | _ => none

@[simp] theorem source_bob_type (high disclose : Bool) :
    sourceBobType (SourcePosition.guessed high disclose).state = some high := by
  cases high <;> cases disclose <;> rfl

theorem source_bob_input (high disclose : Bool) :
    setup.protocolObserve bob (SourcePosition.guessed high disclose).state =
      sourceBobInput disclose := by
  change (some (.inr (.inr (.inr (.inl ((sourceAliceDone high disclose).view bob))))) :
      setup.ProtocolView bob) =
    some (.inr (.inr (.inr (.inl ((sourceAliceDone true disclose).view bob)))))
  cases high with
  | false =>
      exact congrArg (fun view => (some (.inr (.inr (.inr (.inl view)))) :
        setup.ProtocolView bob))
        (source_bob_view_type_eq disclose).symm
  | true => rfl

theorem source_bob_input_false_ne_true : sourceBobInput false ≠ sourceBobInput true := by
  intro same
  change (some (.inr (.inr (.inr (.inl ((sourceAliceDone true false).view bob))))) :
      setup.ProtocolView bob) =
    some (.inr (.inr (.inr (.inl ((sourceAliceDone true true).view bob))))) at same
  have views := Sum.inl.inj (Sum.inr.inj (Sum.inr.inj (Sum.inr.inj (Option.some.inj same))))
  have publication := congrArg (fun view => view.1.cells.get .here) views
  change (PublicationResult.failure : PublicationResult (Val unitPayload)) =
    .success unitValue at publication
  cases publication

@[simp] theorem source_bob_input_eq_iff (high actual disclose : Bool) :
    setup.protocolObserve bob (SourcePosition.guessed high actual).state =
      sourceBobInput disclose ↔ actual = disclose := by
  rw [source_bob_input]
  cases actual <;> cases disclose
  · simp only [iff_self]
  · exact iff_of_false source_bob_input_false_ne_true Bool.false_ne_true
  · exact iff_of_false (Ne.symm source_bob_input_false_ne_true) (by decide)
  · simp only [iff_self]

@[simp] theorem source_bob_inputs_eq_iff (left right : Bool) :
    sourceBobInput left = sourceBobInput right ↔ left = right := by
  cases left <;> cases right
  · simp only [iff_self]
  · exact iff_of_false source_bob_input_false_ne_true Bool.false_ne_true
  · exact iff_of_false (Ne.symm source_bob_input_false_ne_true) (by decide)
  · simp only [iff_self]

theorem source_history_observe (who : Player) (history : sourceArena.History) :
    sourceModel.infoOf who history.trace = setup.protocolObserve who history.state :=
  setup.protocol_info sourceAdmission who history.trace

/-- Every genuine Bob site carries exactly one of the two actual public
publication inputs. His private observation adds no hidden-type distinction. -/
theorem source_bob_site_input (site : sourceModel.InformationSite bob) :
    ∃ disclose, site.1 = sourceBobInput disclose := by
  obtain ⟨history, _, _⟩ := site.2
  obtain ⟨high, disclose, same⟩ := source_bob_active_state site history
  refine ⟨disclose, ?_⟩
  calc
    site.1 = sourceModel.infoOf bob history.1.trace := history.2.symm
    _ = setup.protocolObserve bob history.1.state := source_history_observe bob history.1
    _ = sourceBobInput disclose := by rw [same, source_bob_input]

/-- The incoming joint mass is the actual prior type mass times Alice's
actual source disclosure atom. Bob's later response law has no effect on it. -/
theorem source_initialized_bob_joint (aliceLaw bobLaw : Bool → PMF Bool)
    (high disclose : Bool) :
    (((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).toOuterMeasure
      {history | sourceModel.infoOf bob history.trace = sourceBobInput disclose ∧
        sourceBobType history.state = some high}).toReal) =
      (if high then (1 / 4 : ℝ) else 3 / 4) * (aliceLaw high disclose).toReal := by
  classical
  let test : setup.ProtocolState → ℝ := fun state =>
    if setup.protocolObserve bob state = sourceBobInput disclose ∧
      sourceBobType state = some high then 1 else 0
  calc
    _ = expect (sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4)
        (fun history => test history.state) := by
      rw [← expect_indicator]
      apply expect_congr_on_support
      intro history _
      simp only [source_history_observe, test, Set.mem_ofPred_eq]
      split_ifs <;> rfl
    _ = expect ((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).map
        ExecutionProtocol.History.state) test := by
      simpa only [Function.comp_def] using
        (expect_map (fun history : sourceArena.History => history.state)
          (sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4) test).symm
    _ = _ := by
      rw [source_initialized_bob_law, expect_mix]
      · rw [expect_map, expect_map]
        cases high <;>
          norm_num [test, Function.comp_def, source_bob_input, source_bob_inputs_eq_iff,
            expect_ite_eq, expect_constant, eq_comm]
      · exact payoffIntegrable_ite_one_zero _ _
      · exact payoffIntegrable_ite_one_zero _ _

/-- The source information denominator is derived from the same actual
initialized execution, not supplied as a posterior premise. -/
theorem source_initialized_bob_mass (aliceLaw bobLaw : Bool → PMF Bool)
    (disclose : Bool) :
    (((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).toOuterMeasure
      {history | sourceModel.infoOf bob history.trace = sourceBobInput disclose}).toReal) =
      (1 / 4 : ℝ) * (aliceLaw true disclose).toReal +
        (3 / 4 : ℝ) * (aliceLaw false disclose).toReal := by
  classical
  let test : setup.ProtocolState → ℝ := fun state =>
    if setup.protocolObserve bob state = sourceBobInput disclose then 1 else 0
  calc
    _ = expect (sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4)
        (fun history => test history.state) := by
      rw [← expect_indicator]
      apply expect_congr_on_support
      intro history _
      simp only [source_history_observe, test, Set.mem_ofPred_eq]
      split_ifs <;> rfl
    _ = expect ((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).map
        ExecutionProtocol.History.state) test := by
      simpa only [Function.comp_def] using
        (expect_map (fun history : sourceArena.History => history.state)
          (sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4) test).symm
    _ = _ := by
      rw [source_initialized_bob_law, expect_mix]
      · rw [expect_map, expect_map]
        norm_num [test, Function.comp_def, source_bob_input, source_bob_inputs_eq_iff,
          expect_ite_eq, expect_constant, eq_comm]
      · exact payoffIntegrable_ite_one_zero _ _
      · exact payoffIntegrable_ite_one_zero _ _

def sourceBayes (aliceLaw bobLaw : Bool → PMF Bool)
    (fullAlice : ∀ high, FullSupport (aliceLaw high))
    (fullBob : ∀ disclose, FullSupport (bobLaw disclose)) :
    sourceModel.BehavioralAssessment :=
  sourceModel.bayesAssessment (sourceProfile aliceLaw bobLaw)
    (fun who site => sourceProfile_full aliceLaw bobLaw fullAlice fullBob who site.1)
    (setup.decision_antichain sourceAdmission)

/-- The Bayes type mass is obtained by conditioning actual source play at
Bob's common depth. Its denominator and numerator are the proved execution
likelihoods, with no belief equation supplied as an assumption. -/
theorem source_bayes_type_mass (aliceLaw bobLaw : Bool → PMF Bool)
    (fullAlice : ∀ high, FullSupport (aliceLaw high))
    (fullBob : ∀ disclose, FullSupport (bobLaw disclose))
    (site : sourceModel.InformationSite bob) (disclose : Bool)
    (observed : site.1 = sourceBobInput disclose) :
    (((sourceBayes aliceLaw bobLaw fullAlice fullBob).belief bob site).toOuterMeasure
      {history | sourceBobType history.1.state = some true}).toReal =
      ((1 / 4 : ℝ) * (aliceLaw true disclose).toReal) /
        ((1 / 4 : ℝ) * (aliceLaw true disclose).toReal +
          (3 / 4 : ℝ) * (aliceLaw false disclose).toReal) := by
  classical
  have mixed : (sourceBayes aliceLaw bobLaw fullAlice fullBob).IsFullyMixed :=
    fun who site => sourceProfile_full aliceLaw bobLaw fullAlice fullBob who site.1
  have meet : ∃ history ∈ {history | sourceModel.infoOf bob history.trace = site.1},
      history ∈ (sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).support := by
    obtain ⟨history, _, _⟩ := site.2
    refine ⟨history.1, history.2, ?_⟩
    have supported := mixed.history_supported history.1.trace
    rwa [source_bob_depth site history] at supported
  have conditioned := sourceModel.bayesBelief_map_eq_filter
    (sourceProfile aliceLaw bobLaw) bob site 4 (source_bob_depth site)
    ((setup.decision_antichain sourceAdmission) bob site)
    (sourceModel.informationMass_pos_of_fullSupport _ mixed bob site) meet
  have events := congrArg (fun law : PMF sourceArena.History =>
    (law.toOuterMeasure {history | sourceBobType history.state = some true}).toReal) conditioned
  rw [PMF.toOuterMeasure_map_apply, pmf_toOuterMeasure_filter_apply,
    ENNReal.toReal_div, Set.inter_comm] at events
  change _ = (((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).toOuterMeasure
      {history | sourceModel.infoOf bob history.trace = site.1 ∧
        sourceBobType history.state = some true}).toReal) /
    (((sourceModel.runBehavioral (sourceProfile aliceLaw bobLaw) 4).toOuterMeasure
      {history | sourceModel.infoOf bob history.trace = site.1}).toReal) at events
  conv_rhs at events =>
    rw [observed, source_initialized_bob_joint, source_initialized_bob_mass]
  exact events
