These marks bound expected terminal liabilities. They do not assert that a
watcher's report has been delivered or that settlement has already collected.

noncomputable section

open Vegas Vegas.EventGraphRuntime Interaction GameTheory.Math.Probability

theorem rejected_alice_persists {before after : List (MessageId Player × Bool)}
    (retained : before <+: after) (rejected : rejectedAlice before = true) :
    rejectedAlice after = true := by
  obtain ⟨tail, rfl⟩ := retained
  change (before ++ tail).any _ = true
  rw [List.any_append]
  change (rejectedAlice before || rejectedAlice tail) = true
  rw [rejected]
  rfl

theorem aliceLiability_policyInvariant (players : Player → nativeApp.Policy) :
    nativeApp.PolicyInvariant players (fun execution => aliceLiability execution = true) where
  respond execution who action liable _ := by
    rcases action with ⟨transmission⟩
    cases transmission <;> exact liable
  environment execution next command liable reached := by
    simp only [aliceLiability, Bool.or_eq_true] at liable ⊢
    rcases liable with rejected | observed
    · exact Or.inl (rejected_alice_persists
        (nativeApp.environmentStep_receipts_prefix execution next command reached) rejected)
    · right
      obtain ⟨message, present, marked⟩ := List.any_eq_true.mp observed
      apply List.any_eq_true.mpr
      exact ⟨message, (nativeApp.leaked_policyInvariant players watcher message).environment
        execution next command present reached, marked⟩

theorem native_plan_liability (players : Player → nativeApp.Policy)
    (plan : List (ServiceInstruction nativeGraph)) (before after : nativeApp.Execution)
    (supported : after ∈
      (nativeRuntime.runInteractionPlan nativeLeaks players nativeNetwork plan before).support) :
    aliceLiability before = true → aliceLiability after = true := by
  intro liable
  induction plan generalizing before with
  | nil => cases (PMF.mem_support_pure_iff _ _).mp supported; exact liable
  | cons instruction rest ih =>
      obtain ⟨middle, reached, moved⟩ :=
        Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ supported)
      obtain ⟨command, _, executed⟩ :=
        Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ reached)
      exact ih middle moved ((aliceLiability_policyInvariant players).dispatch
        command before middle liable executed)

theorem native_dispatch_watcher_leaked (players : Player → nativeApp.Policy)
    (command : nativeApp.Command) (before after : nativeApp.Execution)
    (inactive : command.actor? nativeApp ≠ some watcher)
    (supported : after ∈ (nativeApp.dispatch players command before).support) :
    after.network.leaked watcher = before.network.leaked watcher := by
  obtain ⟨middle, moved, resumed⟩ :=
    Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ supported)
  have environment : middle.network.leaked watcher = before.network.leaked watcher := by
    cases command with
    | activate who =>
        rw [ReactiveApplication.Execution.activation_samples, PMF.support_map] at moved
        obtain ⟨selected, _, rfl⟩ := moved
        change (before.network.learn who selected).leaked watcher = before.network.leaked watcher
        exact before.network.learn_other who watcher selected (fun same => inactive (same ▸ rfl))
    | wait =>
        simp only [ReactiveApplication.Execution.environmentStep, PMF.pure_map] at moved
        cases (PMF.mem_support_pure_iff _ _).mp moved
        rfl
    | «include» id =>
        simp only [ReactiveApplication.Execution.environmentStep, PMF.pure_map] at moved
        cases (PMF.mem_support_pure_iff _ _).mp moved
        rw [nativeApp.includePending_network]
        unfold MessageNetwork.includePending
        split <;> rfl
    | application command =>
        obtain ⟨updated, updatedMem, rfl⟩ := PMF.support_map .. ▸ moved
        obtain ⟨state, _, rfl⟩ := PMF.support_map .. ▸ updatedMem
        rfl
  cases actor : command.actor? nativeApp with
  | none =>
      simp only [ReactiveApplication.resume, actor, PMF.mem_support_pure_iff _ _] at resumed
      subst after
      exact environment
  | some who =>
      simp only [ReactiveApplication.resume, actor, ReactiveApplication.invoke,
        PMF.support_map] at resumed
      obtain ⟨action, _, rfl⟩ := resumed
      rw [nativeApp.respond_leaked]
      exact environment

theorem native_plan_watcher_leaked (players : Player → nativeApp.Policy)
    (plan : List (ServiceInstruction nativeGraph)) (before after : nativeApp.Execution)
    (inactive : ∀ instruction ∈ plan, instruction ≠ .player watcher)
    (supported : after ∈
      (nativeRuntime.runInteractionPlan nativeLeaks players nativeNetwork plan before).support) :
    after.network.leaked watcher = before.network.leaked watcher := by
  induction plan generalizing before with
  | nil => cases (PMF.mem_support_pure_iff _ _).mp supported; rfl
  | cons instruction rest ih =>
      obtain ⟨middle, reached, moved⟩ :=
        Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ supported)
      obtain ⟨command, scheduled, executed⟩ :=
        Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ reached)
      have actor : command.actor? nativeApp ≠ some watcher := by
        cases instruction with
        | player who =>
            cases (PMF.mem_support_pure_iff _ _).mp scheduled
            intro same
            apply inactive (.player who) (List.mem_cons_self ..)
            exact congrArg ServiceInstruction.player (Option.some.inj same)
        | wire =>
            simp only [interactionInstruction, nativeNetwork, PMF.pure_map,
              PMF.mem_support_pure_iff _ _] at scheduled
            subst command
            simp [NetworkChoice.command, ReactiveApplication.atMostOnceCommand,
              ReactiveApplication.Command.actor?]
        | includeLatest event who =>
            cases (PMF.mem_support_pure_iff _ _).mp scheduled
            unfold reactiveLatest
            split <;> simp [ReactiveApplication.Command.actor?]
        | sample event | tick | expire event =>
            cases (PMF.mem_support_pure_iff _ _).mp scheduled
            simp [ReactiveApplication.Command.actor?]
      exact (ih middle (fun item member => inactive item (List.mem_cons_of_mem _ member))
        moved).trans (native_dispatch_watcher_leaked players command before middle actor executed)
