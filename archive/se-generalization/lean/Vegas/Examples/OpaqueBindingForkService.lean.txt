/-! # A public two-player builder with an opaque binding timing fork

The first two samples make Alice's binding deadline three. The builder fairly
exchanges one tick and her second activation, then includes the same bare call
at clock one. Bob's subsequent resolution is serviced in both branches.
-/

noncomputable section

namespace Vegas.OpaqueBindingFork

open SourceProgram EventGraph EventGraphRuntime Interaction GameTheory.Math.Probability

abbrev Player := Fin 2
abbrev alice : Player := 0
abbrev bob : Player := 1
abbrev initialCtx : SourceCtx Player simpleExpr := [(0, .commitment bob .bool)]

private def acceptGuard {Γ : SourceCtx Player simpleExpr} (who : Player) (name : VarId) :
    SourceGuard simpleExpr Γ who name .bool where
  schema := []
  schemaNames := by decide
  subjectFresh := by simp
  code := .constBool true
  reads := fun h => nomatch h

def program : SourceProgram Player simpleExpr initialCtx {0} :=
  .sample 1 (payload := .bool) (by decide) (.weighted (.pure true)) <|
  .sample 2 (payload := .bool) (by decide) (.weighted (.pure true)) <|
  .commit 3 alice (by decide) (acceptGuard alice 3) <|
  .reveal 4 bob 0 (by decide) (.there (.there (.there .here))) (by decide) <|
  .reveal 5 alice 3 (by decide) (.there .here) (by decide) <|
  .ret [(alice, .ite (.isSuccess (.var 5 .here)) (.constInt 1) (.constInt 0))]

def sourceInitial : State simpleExpr initialCtx := Env.cons (.success true) (Env.empty _)

def setup : Setup (Player := Player) (L := simpleExpr) where
  context := initialCtx
  namesNodup := by decide
  initialLaw := PMF.pure sourceInitial
  obligations := {0}
  program := program
  accounts := rfl

abbrev nativeGraph := graph setup
abbrev sample0 : nativeGraph.EventId := ⟨0, by decide⟩
abbrev sample1 : nativeGraph.EventId := ⟨1, by decide⟩
abbrev binding : nativeGraph.EventId := ⟨2, by decide⟩
abbrev bobResolution : nativeGraph.EventId := ⟨3, by decide⟩
abbrev aliceResolution : nativeGraph.EventId := ⟨4, by decide⟩

def leaks : MessageNetwork.ObservationRule Player (WitnessedPacket nativeGraph) :=
  fun _ _ => PMF.pure ∅

abbrev app := application setup leaks

def lastTick (history : List app.EnvironmentEntry) : Bool :=
  history.getLast?.any (fun entry => match entry.command with
    | .application .advanceClock => true
    | _ => false)

def stageCommand (stage : Nat) (history : List app.EnvironmentEntry)
    (view : app.EnvironmentView) : app.Command :=
  match stage with
  | 0 => .application (.executeSample sample0)
  | 1 => .application (.executeSample sample1)
  | 2 => .activate alice
  | 3 | 6 => (runtime setup).reactiveLatest leaks binding alice view
  | 5 => if lastTick history then .activate alice else .application .advanceClock
  | 7 | 8 => .application .advanceClock
  | 9 => .application (.expire binding)
  | 10 => .activate bob
  | 11 => (runtime setup).reactiveLatest leaks bobResolution bob view
  | 12 | 13 | 14 | 15 => .application .advanceClock
  | 16 => .application (.expire bobResolution)
  | 17 => .activate alice
  | 18 => (runtime setup).reactiveLatest leaks aliceResolution alice view
  | 19 | 20 | 21 | 22 | 23 => .application .advanceClock
  | 24 => .application (.expire aliceResolution)
  | _ => .wait

def scheduler : app.Scheduler := fun history view =>
  if history.length = 4 then mix (1 / 2) (by norm_num) (by norm_num)
    (PMF.pure (.activate alice)) (PMF.pure (.application .advanceClock))
  else PMF.pure (stageCommand history.length history view)

def horizon : Nat := 25

def delay (event : nativeGraph.EventId) : Nat :=
  if event = bobResolution then 3 else if event = aliceResolution then 4 else 0

def bound (event : nativeGraph.EventId) : Nat := if event = binding then 2 else 0

@[simp] theorem sample0_actor : nativeGraph.actor? sample0 = none := rfl
@[simp] theorem sample1_actor : nativeGraph.actor? sample1 = none := rfl
@[simp] theorem binding_actor : nativeGraph.actor? binding = some alice := rfl
@[simp] theorem bobResolution_actor : nativeGraph.actor? bobResolution = some bob := rfl
@[simp] theorem aliceResolution_actor : nativeGraph.actor? aliceResolution = some alice := rfl

theorem timely : AsyncTimely (runtime setup) delay bound := by
  intro event owned
  change Fin 5 at event
  fin_cases event
  · change false = true at owned
    contradiction
  · change false = true at owned
    contradiction
  · change 0 + 2 < 3
    decide
  · change 3 + 0 < 4
    decide
  · change 4 + 0 < 5
    decide

def stageClock (history : List app.EnvironmentEntry) : Nat :=
  let stage := history.length
  if stage ≤ 4 then 0
  else if stage = 5 then if lastTick history then 1 else 0
  else if stage ≤ 7 then 1
  else if stage ≤ 8 then 2
  else if stage ≤ 12 then 3
  else if stage ≤ 16 then stage - 9
  else if stage ≤ 19 then 7
  else if stage ≤ 24 then stage - 12 else 12

def lowerRank (stage : Nat) : Nat :=
  if stage = 0 then 0 else if stage = 1 then 1
  else if stage ≤ 9 then 2 else if stage ≤ 16 then 3
  else if stage ≤ 24 then 4 else 5

def upperRank (stage : Nat) : Nat :=
  if stage ≤ 1 then stage else if stage ≤ 9 then 3
  else if stage ≤ 16 then 4 else 5

private theorem environment_effect (execution next : app.Execution) (command : app.Command)
    (moved : next ∈ (execution.environmentStep app command).support) :
    (command = .application .advanceClock ∧
      next.application.config = execution.application.config ∧
      next.application.activatedAt = execution.application.activatedAt ∧
      next.application.clock = execution.application.clock + 1) ∨
      GraphStep execution.application next.application := by
  cases command with
  | activate who =>
      obtain ⟨updated, supported, rfl⟩ := PMF.support_map .. ▸ moved
      obtain ⟨selected, _, rfl⟩ := PMF.support_map .. ▸ supported
      exact Or.inr (GraphStep.refl _)
  | wait =>
      simp only [ReactiveApplication.Execution.environmentStep, PMF.pure_map] at moved
      cases (PMF.mem_support_pure_iff _ _).mp moved
      exact Or.inr (GraphStep.refl _)
  | «include» id =>
      simp only [ReactiveApplication.Execution.environmentStep, PMF.pure_map] at moved
      cases (PMF.mem_support_pure_iff _ _).mp moved
      exact Or.inr (graphStep_includePending (runtime setup) leaks execution id)
  | application command =>
      have supported := (applicationStep_facts execution next command moved).1
      cases command with
      | advanceClock =>
          simp only [environmentStep, PMF.mem_support_pure_iff _ _] at supported
          rw [supported]
          exact Or.inl ⟨rfl, rfl, rfl, rfl⟩
      | executeSample event =>
          exact Or.inr (graphStep_executeSample (runtime setup) _ _ event supported)
      | expire event =>
          exact Or.inr (graphStep_expire (runtime setup) _ _ event supported)

private def tickCount : app.Command → Nat
  | .application .advanceClock => 1
  | _ => 0

private theorem selected_four (history : List app.EnvironmentEntry) (view : app.EnvironmentView)
    (command : app.Command) (four : history.length = 4)
    (selected : command ∈ (scheduler history view).support) :
    command = .activate alice ∨ command = .application .advanceClock := by
  simp only [scheduler, four, ite_true] at selected
  exact (mem_support_mix_pure_iff _ _ _ (by norm_num) (by norm_num)).mp selected

private theorem selected_other (history : List app.EnvironmentEntry) (view : app.EnvironmentView)
    (command : app.Command) (other : history.length ≠ 4)
    (selected : command ∈ (scheduler history view).support) :
    command = stageCommand history.length history view := by
  simp only [scheduler, other, ite_false, PMF.mem_support_pure_iff] at selected
  exact selected

private theorem environment_clock (execution next : app.Execution) (command : app.Command)
    (moved : next ∈ (execution.environmentStep app command).support) :
    next.application.clock = execution.application.clock + tickCount command := by
  rcases environment_effect execution next command moved with ⟨tick, _, _, clock⟩ | step
  · rw [tick, clock]
    rfl
  · have clock : next.application.clock = execution.application.clock := by
      rcases step with ⟨_, same, _⟩ | ⟨_, _, _, _, same, _⟩ <;> exact same
    cases command with
    | application command =>
        cases command with
        | advanceClock =>
            have physical := (applicationStep_facts execution next _ moved).1
            simp only [environmentStep, PMF.mem_support_pure_iff _ _] at physical
            rw [physical] at clock
            change execution.application.clock + 1 = execution.application.clock at clock
            omega
        | executeSample | expire => exact clock
    | activate | wait | «include» => exact clock

private theorem stageClock_append (history : List app.EnvironmentEntry)
    (view : app.EnvironmentView) (command : app.Command) (beforeEnd : history.length < horizon)
    (selected : command ∈ (scheduler history view).support) :
    stageClock (history ++ [⟨view, command⟩]) = stageClock history + tickCount command := by
  by_cases four : history.length = 4
  · rcases selected_four history view command four selected with rfl | rfl <;>
      simp [stageClock, four, lastTick, tickCount]
  have chosen := selected_other history view command four selected
  rw [chosen]
  generalize stageEq : history.length = stage at beforeEnd four ⊢
  change stage < 25 at beforeEnd
  interval_cases stage
  all_goals try contradiction
  all_goals simp only [stageCommand]
  all_goals first
    | (unfold reactiveLatest; split <;> simp [stageClock, stageEq, tickCount])
    | (split <;> simp_all [stageClock, lastTick, tickCount])
    | simp [stageClock, stageEq, tickCount]

private theorem selected_activation (history : List app.EnvironmentEntry)
    (view : app.EnvironmentView) (command : app.Command) (beforeEnd : history.length < horizon)
    (selected : command ∈ (scheduler history view).support) (who : Player)
    (active : command.actor? app = some who) :
    (history.length + 1 = 3 ∧ who = alice) ∨
      (history.length + 1 = 5 ∧ who = alice) ∨
      (history.length + 1 = 6 ∧ who = alice) ∨
      (history.length + 1 = 11 ∧ who = bob) ∨
      (history.length + 1 = 18 ∧ who = alice) := by
  by_cases four : history.length = 4
  · rcases selected_four history view command four selected with rfl | rfl
    · have same : who = alice := (Option.some.inj active).symm
      exact Or.inr (Or.inl ⟨by omega, same⟩)
    · cases active
  have chosen := selected_other history view command four selected
  rw [chosen] at active
  generalize stageEq : history.length = stage at beforeEnd four active ⊢
  change stage < 25 at beforeEnd
  interval_cases stage
  all_goals simp only [stageCommand] at active
  all_goals first
    | (unfold reactiveLatest at active; split at active <;> cases active)
    | (split at active <;> simp_all [ReactiveApplication.Command.actor?])
    | simp_all [ReactiveApplication.Command.actor?]

def opportunityEnd (event : nativeGraph.EventId) : Nat :=
  if event = binding then 3 else if event = bobResolution then 11 else 18

def inclusionEnd (event : nativeGraph.EventId) : Nat :=
  if event = binding then 7 else if event = bobResolution then 12 else 19

def ActivationBounds (event : nativeGraph.EventId) (entered : Nat) : Prop :=
  if event = binding then entered = 0
  else if event = bobResolution then entered ≤ 3
  else if event = aliceResolution then 3 ≤ entered ∧ entered ≤ 7 else False

structure Phase (control : app.Control) : Prop where
  clock : control.execution.application.clock = stageClock control.execution.environmentRecall
  ordered : ∃ rank, lowerRank control.execution.environmentRecall.length ≤ rank ∧
    rank ≤ upperRank control.execution.environmentRecall.length ∧
    control.execution.application.config.cut.IsPrefix rank
  entered : ∀ event entered, control.execution.application.activatedAt event = some entered →
    ActivationBounds event entered
  opportunity : ∀ event who entered, nativeGraph.actor? event = some who →
    control.execution.application.config.cut.Ready event →
    control.execution.application.activatedAt event = some entered →
    opportunityEnd event ≤ control.execution.environmentRecall.length →
    OwnerActivatedSince (runtime setup) leaks control.execution.environmentRecall event who entered
  activation : ∀ who, control.actor = some who →
    (control.execution.environmentRecall.length = 3 ∧ who = alice) ∨
      (control.execution.environmentRecall.length = 5 ∧ who = alice) ∨
      (control.execution.environmentRecall.length = 6 ∧ who = alice) ∨
      (control.execution.environmentRecall.length = 11 ∧ who = bob) ∨
      (control.execution.environmentRecall.length = 18 ∧ who = alice)
  responses : ∀ who, ∀ entry ∈ control.execution.recall who,
    (who = alice ∧ (entry.beforeView.application.publicView.clock = 0 ∨
      entry.beforeView.application.publicView.clock = 1 ∨
      entry.beforeView.application.publicView.clock = 7)) ∨
    (who = bob ∧ entry.beforeView.application.publicView.clock = 3)
  lateResponses : ∀ entry ∈ control.execution.recall alice,
    entry.beforeView.application.publicView.EventReady aliceResolution →
    entry.beforeView.application.publicView.clock = 7
  inclusion : ∀ event who, nativeGraph.actor? event = some who →
    inclusionEnd event ≤ control.execution.environmentRecall.length →
    event ∉ control.execution.application.config.cut.completed →
    ∀ earlier later entry message, control.execution.recall who = earlier ++ entry :: later →
    entry.emitted = some message → message.sender = who →
    message.payload.call.event? nativeGraph = some event →
    entry.beforeView.application.publicView.EventReady event →
    (∀ other ∈ earlier ++ later, ¬ EmitsOtherFor (runtime setup) leaks other event message.id) →
    ∃ accepted, (message.id, accepted) ∈ control.execution.receipts

def phaseInvariant : app.ProtocolState → Prop
  | none => True
  | some control => Phase control

theorem phase_initial (state : app.State) (supported : state ∈ (initialLaw setup).support) :
    Phase ⟨horizon, none, ReactiveApplication.Execution.initial app state⟩ := by
  obtain ⟨initial, selected, rfl⟩ := PMF.support_map .. ▸ supported
  have same := (PMF.mem_support_pure_iff _ _).mp selected
  subst initial
  have ordered : (EventGraphRuntime.State.initial
      (setup.eventInputs sourceInitial)).config.cut.IsPrefix 0 := EventOrder.Cut.empty_isPrefix _
  refine {
    clock := rfl
    ordered := ⟨0, le_rfl, le_rfl, ordered⟩
    entered := ?_
    opportunity := ?_
    activation := by simp
    responses := by simp [ReactiveApplication.Execution.initial]
    lateResponses := by simp [ReactiveApplication.Execution.initial]
    inclusion := ?_ }
  · intro event entered activated
    let initialState : EventGraphRuntime.State nativeGraph :=
      EventGraphRuntime.State.initial (setup.eventInputs sourceInitial)
    change initialState.activatedAt event = some entered at activated
    have invariant := EventGraphRuntime.State.initial_invariant (graph := nativeGraph)
      (setup.eventInputs sourceInitial)
    have strategic := invariant.activated_iff event |>.mp (by rw [activated]; rfl)
    have rank := (ready_iff_rank setup initialState.config 0 ordered event).mp strategic.1
    have eventEq : event = sample0 := Fin.ext rank
    rw [eventEq, sample0_actor] at strategic
    cases strategic.2
  · intro event who entered owned _ready _activated visited
    have positive : 0 < opportunityEnd event := by
      unfold opportunityEnd
      split_ifs <;> omega
    change opportunityEnd event ≤ 0 at visited
    omega
  · intro event who owned included
    have positive : 0 < inclusionEnd event := by
      unfold inclusionEnd
      split_ifs <;> omega
    change inclusionEnd event ≤ 0 at included
    omega

private theorem active_after_inclusion_impossible (control : app.Control) (phase : Phase control)
    (who : Player) (active : control.actor = some who) (event : nativeGraph.EventId)
    (owned : nativeGraph.actor? event = some who)
    (ready : control.execution.application.config.cut.Ready event)
    (included : inclusionEnd event ≤ control.execution.environmentRecall.length) : False := by
  obtain ⟨rank, lower, _upper, ordered⟩ := phase.ordered
  have current := (ready_iff_rank setup _ rank ordered event).mp ready
  have activation := phase.activation who active
  fin_cases event
  · change none = some who at owned
    cases owned
  · change none = some who at owned
    cases owned
  · have same : who = alice := (Option.some.inj owned).symm
    subst who
    change 7 ≤ control.execution.environmentRecall.length at included
    rcases activation with ⟨first, _⟩ | ⟨second, _⟩ | ⟨third, _⟩ | ⟨fourth, same⟩ | ⟨last, _⟩
    · omega
    · omega
    · omega
    · cases same
    · rw [last] at lower
      change 4 ≤ rank at lower
      change 2 = rank at current
      omega
  · have same : who = bob := (Option.some.inj owned).symm
    subst who
    change 12 ≤ control.execution.environmentRecall.length at included
    rcases activation with ⟨_, same⟩ | ⟨_, same⟩ | ⟨_, same⟩ | ⟨fourth, _⟩ | ⟨_, same⟩
    all_goals first | cases same | omega
  · have same : who = alice := (Option.some.inj owned).symm
    subst who
    change 19 ≤ control.execution.environmentRecall.length at included
    rcases activation with ⟨first, _⟩ | ⟨second, _⟩ | ⟨third, _⟩ | ⟨fourth, same⟩ | ⟨last, _⟩
    · omega
    · omega
    · omega
    · cases same
    · omega

theorem phase_respond (control : app.Control) (phase : Phase control)
    (who : Player) (active : control.actor = some who) (response : app.Action) :
    Phase { control with
      actor := none
      execution := control.execution.respond app who response } := by
  obtain ⟨configEq, publicEq⟩ :=
    (runtime setup).reactive_respond_application leaks control.execution who response
  have clockEq := congrArg PublicView.clock publicEq
  have activatedEq := congrArg PublicView.activatedAt publicEq
  dsimp only [EventGraphRuntime.State.publicView] at clockEq activatedEq
  have recallEq := app.respond_environmentRecall control.execution who response
  have receiptEq := app.respond_receipts control.execution who response
  refine {
    clock := by rw [clockEq, recallEq]; exact phase.clock
    ordered := by rw [configEq, recallEq]; exact phase.ordered
    entered := by rw [activatedEq]; exact phase.entered
    opportunity := by rw [configEq, activatedEq, recallEq]; exact phase.opportunity
    activation := by intro observer impossible; cases impossible
    responses := ?_
    lateResponses := ?_
    inclusion := ?_ }
  · intro observer entry member
    rcases app.respond_entry_origin control.execution who observer response entry member with
      prior | ⟨same, fresh⟩
    · exact phase.responses observer entry prior
    · subst observer
      rw [fresh]
      change (who = alice ∧ (control.execution.application.clock = 0 ∨
        control.execution.application.clock = 1 ∨ control.execution.application.clock = 7)) ∨
        (who = bob ∧ control.execution.application.clock = 3)
      rw [phase.clock]
      rcases phase.activation who active with ⟨first, owner⟩ | ⟨second, owner⟩ |
          ⟨third, owner⟩ | ⟨fourth, owner⟩ | ⟨last, owner⟩
      · exact Or.inl ⟨owner, by simp [stageClock, first]⟩
      · exact Or.inl ⟨owner, by simp [stageClock, second]; split <;> simp⟩
      · exact Or.inl ⟨owner, by simp [stageClock, third]⟩
      · exact Or.inr ⟨owner, by simp [stageClock, fourth]⟩
      · exact Or.inl ⟨owner, by simp [stageClock, last]⟩
  · intro entry member ready
    rcases app.respond_entry_origin control.execution who alice response entry member with
      prior | ⟨same, fresh⟩
    · exact phase.lateResponses entry prior ready
    · rw [fresh] at ready ⊢
      have owner : who = alice := same.symm
      have actualReady := (control.execution.application.publicView_eventReady
        aliceResolution).mp ready
      obtain ⟨rank, _, high, ordered⟩ := phase.ordered
      have current := (ready_iff_rank setup _ rank ordered aliceResolution).mp actualReady
      change 4 = rank at current
      rw [← current] at high
      rcases phase.activation who active with ⟨first, _⟩ | ⟨second, _⟩ |
          ⟨third, _⟩ | ⟨_, bobOwner⟩ | ⟨last, _⟩
      · simp only [first, upperRank, Nat.reduceLeDiff, ite_false, ite_true] at high
      · simp only [second, upperRank, Nat.reduceLeDiff, ite_false, ite_true] at high
      · simp only [third, upperRank, Nat.reduceLeDiff, ite_false, ite_true] at high
      · have impossible : alice = bob := owner.symm.trans bobOwner
        cases impossible
      · change control.execution.application.clock = 7
        rw [phase.clock]
        simp [stageClock, last]
  · intro event owner owned included unfinished earlier later entry message split emitted authored
      addressed ready sole
    rw [recallEq] at included
    rw [configEq] at unfinished
    rw [receiptEq]
    by_cases same : who = owner
    · rw [← same] at owned split authored
      change (control.execution.respond app who response).recall who = earlier ++ entry :: later
        at split
      obtain ⟨output, afterRecall, _identity⟩ :=
        respond_recall_self setup leaks control.execution who response
      rw [afterRecall] at split
      rcases List.eq_nil_or_concat later with rfl | ⟨init, last, rfl⟩
      · obtain ⟨_, entryEq⟩ := List.append_inj' split rfl
        have fresh := (List.singleton_inj.mp entryEq).symm
        subst entry
        exact (active_after_inclusion_impossible control phase who active event owned
          ((control.execution.application.publicView_eventReady event).mp ready) included).elim
      · simp only [List.concat_eq_append] at split sole
        have reassociated : earlier ++ entry :: (init ++ [last]) =
            (earlier ++ entry :: init) ++ [last] := by simp
        rw [reassociated] at split
        obtain ⟨beforeRecall, _⟩ := List.append_inj' split rfl
        exact phase.inclusion event who owned included unfinished earlier init entry message
          beforeRecall emitted authored addressed ready (by
            intro other member
            apply sole other
            simp only [List.mem_append] at member ⊢
            tauto)
    · change (control.execution.respond app who response).recall owner = earlier ++ entry :: later
        at split
      rw [app.respond_recall_other control.execution who owner (Ne.symm same) response] at split
      exact phase.inclusion event owner owned included unfinished earlier later entry message split
        emitted authored addressed ready sole

private theorem Phase.window (control : app.Control) (phase : Phase control)
    (event : nativeGraph.EventId)
    (lower : lowerRank control.execution.environmentRecall.length = event.val)
    (upper : upperRank control.execution.environmentRecall.length = event.val + 1) :
    control.execution.application.config.cut.IsPrefix event.val ∨
      control.execution.application.config.cut.IsPrefix (event.val + 1) := by
  obtain ⟨rank, low, high, ordered⟩ := phase.ordered
  rw [lower] at low
  rw [upper] at high
  have choice : rank = event.val ∨ rank = event.val + 1 := by omega
  rcases choice with rfl | rfl
  · exact Or.inl ordered
  · exact Or.inr ordered

private theorem phase_environment_ordered (control : app.Control) (phase : Phase control)
    (trace : (app.protocol (initialLaw setup) horizon scheduler).Trace (some control))
    (next : app.Execution) (command : app.Command)
    (selected : command ∈ (scheduler control.execution.environmentRecall
      (control.execution.observeEnvironment app)).support)
    (moved : next ∈ (control.execution.environmentStep app command).support)
    (beforeEnd : control.execution.environmentRecall.length < horizon) :
    ∃ rank, lowerRank next.environmentRecall.length ≤ rank ∧
      rank ≤ upperRank next.environmentRecall.length ∧
      next.application.config.cut.IsPrefix rank := by
  have appended := environmentStep_recall_append control.execution next command moved
  have length : next.environmentRecall.length = control.execution.environmentRecall.length + 1 := by
    rw [appended, List.length_append, List.length_singleton]
  obtain ⟨inputs, invariant⟩ := (roster_trace_facts setup leaks horizon scheduler trace).1
  have unique := app.uniqueIds_history scheduler (initialLaw setup) horizon control trace
  by_cases four : control.execution.environmentRecall.length = 4
  · have same := environmentStep_config_activated_stutter setup leaks
      control.execution next command (by
        rcases selected_four _ _ _ four selected with first | second
        · exact Or.inl ⟨alice, first⟩
        · exact Or.inr (Or.inr second)) moved
    obtain ⟨rank, low, high, ordered⟩ := phase.ordered
    rw [same.1]
    refine ⟨rank, ?_, ?_, ordered⟩
    · simpa [length, four, lowerRank] using low
    · simpa [length, four, upperRank] using high
  have chosen := selected_other _ _ _ four selected
  rw [chosen] at moved
  generalize stageEq : control.execution.environmentRecall.length = stage at beforeEnd four moved ⊢
  change stage < 25 at beforeEnd
  interval_cases stage
  all_goals try contradiction
  case «0» =>
    obtain ⟨rank, low, high, ordered⟩ := phase.ordered
    rw [stageEq] at low high
    have zero : rank = 0 := by simpa [lowerRank, upperRank] using high
    subst rank
    have ready := (ready_iff_rank setup _ 0 ordered sample0).mpr rfl
    have physical := (applicationStep_facts control.execution next _ moved).1
    have completed := (executeSample_completes (runtime setup) _ _ sample0 sample0_actor
      ready physical).1
    refine ⟨1, by simp [length, stageEq, lowerRank], by simp [length, stageEq, upperRank], ?_⟩
    rw [completed]
    exact ordered.complete_at sample0 ready rfl
  case «1» =>
    obtain ⟨rank, low, high, ordered⟩ := phase.ordered
    rw [stageEq] at low high
    have one : rank = 1 := by simpa [lowerRank, upperRank] using Nat.le_antisymm high low
    subst rank
    have ready := (ready_iff_rank setup _ 1 ordered sample1).mpr rfl
    have physical := (applicationStep_facts control.execution next _ moved).1

private theorem phase_environment_entered (control : app.Control) (phase : Phase control)
    (trace : (app.protocol (initialLaw setup) horizon scheduler).Trace (some control))
    (next : app.Execution) (command : app.Command)
    (selected : command ∈ (scheduler control.execution.environmentRecall
      (control.execution.observeEnvironment app)).support)
    (moved : next ∈ (control.execution.environmentStep app command).support)
    (beforeEnd : control.execution.environmentRecall.length < horizon)
    (query : nativeGraph.EventId) (entered : Nat)
    (activated : next.application.activatedAt query = some entered) :
    ActivationBounds query entered := by
  obtain ⟨inputs, invariant⟩ := (roster_trace_facts setup leaks horizon scheduler trace).1
  rcases environment_effect control.execution next command moved with
    ⟨_, _, same, _⟩ | graphStep
  · rw [same] at activated
    exact phase.entered query entered activated
  rcases graphStep with ⟨_, _, same⟩ | ⟨event, ready, action, supported, _, refreshed⟩
  · rw [same] at activated
    exact phase.entered query entered activated
  obtain ⟨rank, low, high, ordered⟩ := phase.ordered
  have eventRank := (ready_iff_rank setup _ rank ordered event).mp ready
  have complete := ordered.complete_at event ready eventRank
  have nextOrdered : next.application.config.cut.IsPrefix (rank + 1) := by
    rw [control.execution.application.config.step_cut event ready action
      next.application.config supported]
    exact complete
  have freshInvariant := invariant.refreshStep event ready action next.application.config supported
  have queryFacts := (freshInvariant.activated_iff query).mp (by
    change (State.refreshActivated next.application.config control.execution.application.clock
      control.execution.application.activatedAt query).isSome = true
    rw [← refreshed, activated]
    rfl)
  have queryReady := queryFacts.1
  have successor := (ready_iff_rank setup _ (rank + 1) nextOrdered query).mp queryReady
  have now : entered = control.execution.application.clock := by
    rw [refreshed] at activated
    exact refreshActivated_successor setup invariant event
      (by simpa only [eventRank] using ordered) next.application.config query
      (by omega) entered activated
  rw [now, phase.clock]
  change Fin 5 at query
  fin_cases query
  · have actor := queryFacts.2
    change false = true at actor
    cases actor
  · have actor := queryFacts.2
    change false = true at actor
    cases actor
  · change 2 = rank + 1 at successor
    have one : rank = 1 := by omega
    rw [one] at low high
    have stage : control.execution.environmentRecall.length = 1 := by
      simp only [lowerRank, upperRank] at low high
      split_ifs at low high <;> omega
    change stageClock control.execution.environmentRecall = 0
    simp [stageClock, stage]
  · change 3 = rank + 1 at successor
    have two : rank = 2 := by omega
    rw [two] at low high
    have early : control.execution.environmentRecall.length ≤ 9 := by
      unfold lowerRank at low
      split_ifs at low <;> omega
    change stageClock control.execution.environmentRecall ≤ 3
    dsimp only [stageClock]
    split_ifs <;> omega

    have three : rank = 3 := by omega
    rw [three] at low high
    have early : control.execution.environmentRecall.length ≤ 16 := by
      unfold lowerRank at low
      split_ifs at low <;> omega
    obtain ⟨nextRank, _, nextHigh, after⟩ :=
      phase_environment_ordered control phase trace next command selected moved beforeEnd
    have queryRank := (ready_iff_rank setup _ nextRank after aliceResolution).mp queryReady
    have late : 9 ≤ control.execution.environmentRecall.length := by
      have appended := environmentStep_recall_append control.execution next command moved
      rw [appended, List.length_append, List.length_singleton] at nextHigh
      change 4 = nextRank at queryRank
      rw [← queryRank] at nextHigh
      unfold upperRank at nextHigh
      split_ifs at nextHigh <;> omega
    change 3 ≤ stageClock control.execution.environmentRecall ∧
      stageClock control.execution.environmentRecall ≤ 7
    dsimp only [stageClock]
    split_ifs <;> omega

theorem phase_environment (control : app.Control) (phase : Phase control)
    (trace : (app.protocol (initialLaw setup) horizon scheduler).Trace (some control))
    (remaining : Nat) (counted : control.remaining = remaining + 1)
    (next : app.Execution) (command : app.Command)
    (selected : command ∈ (scheduler control.execution.environmentRecall
      (control.execution.observeEnvironment app)).support)
    (moved : next ∈ (control.execution.environmentStep app command).support) :
    Phase ⟨remaining, command.actor? app, next⟩ := by
  have appended := environmentStep_recall_append control.execution next command moved
  have length : next.environmentRecall.length = control.execution.environmentRecall.length + 1 := by
    rw [appended, List.length_append, List.length_singleton]
  have beforeEnd : control.execution.environmentRecall.length < horizon := by
    have budget := app.raw_trace_accounted (initialLaw setup) horizon scheduler trace
    change control.execution.environmentRecall.length + control.remaining = horizon at budget
    rw [counted] at budget
    omega
  have recallEq := app.environmentStep_recall control.execution next command moved
  refine {
    clock := ?_
    ordered := phase_environment_ordered control phase trace next command selected moved beforeEnd
    entered := phase_environment_entered control phase trace next command selected moved beforeEnd
    opportunity := phase_environment_opportunity control phase next command selected moved
    activation := ?_
    responses := by rw [recallEq]; exact phase.responses
    lateResponses := by rw [recallEq]; exact phase.lateResponses

  · rw [appended, stageClock_append _ _ command beforeEnd selected,
      environment_clock control.execution next command moved, phase.clock]
  · intro who active
    have moments := selected_activation _ _ command beforeEnd selected who active
    simpa only [length] using moments

theorem phase_transition (before after : app.ProtocolState)
    (trace : (app.protocol (initialLaw setup) horizon scheduler).Trace before)
    (valid : phaseInvariant before) (joint : Player → Option app.Action)
    (reached : after ∈ (app.transition (initialLaw setup) horizon scheduler before joint).support) :
    phaseInvariant after := by
  cases before with
  | none =>
      obtain ⟨state, supported, rfl⟩ := PMF.support_map .. ▸ reached
      exact phase_initial state supported
  | some control =>
      rcases control with ⟨remaining, actor, execution⟩
      cases actor with
      | some who =>
          cases (PMF.mem_support_pure_iff _ _).mp reached
          exact phase_respond _ valid who rfl _
      | none =>
          cases remaining with
          | zero =>
              cases (PMF.mem_support_pure_iff _ _).mp reached
              exact valid
          | succ remaining =>
              obtain ⟨command, selected, realized⟩ :=
                Set.mem_iUnion₂.mp (PMF.support_bind .. ▸ reached)
              obtain ⟨next, moved, rfl⟩ := PMF.support_map .. ▸ realized
              exact phase_environment _ valid trace remaining rfl next command selected moved

theorem phase_history : ∀ {state}
    (_trace : (app.protocol (initialLaw setup) horizon scheduler).Trace state), phaseInvariant state
  | _, .start => trivial
  | _, .extend prior joint _ reached =>
      phase_transition _ _ prior (phase_history prior) joint reached
