import Vegas.Game.SourceServiceBindingAttemptLaw
import GameTheory.Math.Probability.ConditionalObservation

/-! # Actual binding receipt conditioning without private-value selection

At one initialized timely canonical first call, conditioning on its actual
acceptance receipt retains the residual source commitment draw. The complete
public record and every foreign input are retained in the same conditional
selection experiment. A missing true receipt selects the actual miss branch
and typed failure. The call may be unprotected.

This uses the physical owner's recorded turn-policy continuation. An arbitrary
free continuation after that call is not covered. The support premise merely
selects a conditioning event of positive actual mass; it supplies no acceptance
probability or source posterior.
-/

noncomputable section

namespace Vegas

open SourceProgram Interaction EventGraphRuntime GameTheory.Math.Probability

variable {Player : Type} [DecidableEq Player]
  {L : IExpr} [IExpr.ResultTypes L]
  {setup : Setup (Player := Player) (L := L)}
  {leaks : MessageNetwork.ObservationRule Player (WitnessedPacket (graph setup))}

/-- The actual receipt-conditioned joint source draw, typed result and full
public/foreign traffic use the same selection kernel for every private value.
Both acceptance and public miss are included. -/
theorem BindingSource.attempt_receipt_posterior
    {Parameter : Type} {horizon remaining : Nat}
    {scheduler : (application setup leaks).Scheduler}
    {delay bound : (graph setup).EventId → Nat}
    (contract : AsyncContract (runtime setup) leaks (initialLaw setup) horizon scheduler
      delay bound)
    (players : Player → (application setup leaks).Policy)
    {turns : Nat} (timing : TurnTiming setup turns) (profile : BehavioralProfile setup.program)
    (execution : (application setup leaks).Execution) (event : (graph setup).EventId)
    (site : BindingSource setup profile event execution.application.config)
    (trace : ((application setup leaks).protocol (initialLaw setup) horizon scheduler).Trace
      (some ⟨remaining, some site.owner, execution⟩))
    (fresh : execution.application.candidates.lookup
      (site.owner, .prepared (execution.application.publicView.bindingCount site.owner)) = .fresh)
    (turn : execution.application.publicView.ownTurn? site.owner = some event)
    (unrecorded : (runtime setup).eventRecorded leaks (execution.recall site.owner) event = false)
    (timely : execution.application.WithinDeadline (runtime setup) event)
    (follows : players site.owner =
      sourceServiceTurnPolicy setup leaks bound turns timing profile site.owner)
    (parameter : Parameter) :
    let serial := execution.application.publicView.bindingCount site.owner
    let draw := commitKernel site.residual (site.source.view site.owner)
    let app := application setup leaks
    let traffic := (runtime setup).bindingPublicTraffic leaks site.owner
    let stop := fun final : app.Execution => event ∈ final.application.config.cut.completed
    let output : EventGraph.FieldRef (graph setup).layout (.binding site.owner site.payload) :=
      ⟨.inr event, site.outputEq⟩
    let actual := draw.bind fun value =>
      (app.runUntilHorizon scheduler players stop horizon
        (execution.respond app site.owner
          ((runtime setup).reactiveBinding leaks site.owner event site.payload value serial))).map
        fun final => (parameter, value, output.get? final.application.config.store, traffic final)
    let reference := (app.runUntilHorizon scheduler players stop horizon
      (execution.respond app site.owner
        ((runtime setup).reactiveBinding leaks site.owner event site.payload .failure serial))).map
          traffic
    ∀ accepted : Bool,
      accepted ∈ (actual.map fun result =>
        decide (((site.owner, execution.network.nextSerial site.owner), true) ∈
          result.2.2.2.2.1)).support →
      fiberPosterior actual (fun result =>
        decide (((site.owner, execution.network.nextSerial site.owner), true) ∈
          result.2.2.2.2.1)) accepted =
      (fiberPosterior reference (fun selected =>
        decide (((site.owner, execution.network.nextSerial site.owner), true) ∈ selected.2.1))
          accepted).bind fun selected => draw.map fun value =>
            (parameter, value, some (if accepted then value else .failure), selected) := by
  classical
  intro serial draw app traffic stop output actual reference accepted possible
  let Traffic := MessageNetwork Player app.Payload × List (MessageId Player × Bool) ×
    List app.EnvironmentEntry × PublicView (graph setup) ×
      (Player → Option (List app.PlayerEntry × PlayerView (graph setup)))
  let Result := Parameter × PublicationResult (L.Val site.payload) ×
    Option (PublicationResult (L.Val site.payload)) × Traffic
  let selected : Traffic → Bool := fun read =>
    decide (((site.owner, execution.network.nextSerial site.owner), true) ∈
      read.2.1)
  let observed : Result → Bool := fun result => selected result.2.2.2
  let kernel : Traffic → PMF Result := fun read => draw.map fun value =>
    (parameter, value, some (if selected read then value else .failure), read)
  have actualLaw : actual = reference.bind kernel := by
    have law := site.attempt_law contract players timing profile execution event trace fresh turn
      unrecorded timely follows parameter
    dsimp only at law
    simpa only [actual, reference, kernel, selected, serial, draw, app, traffic, stop, output,
      decide_eq_true_eq] using law
  have determines : ∀ read ∈ reference.support, ∀ result ∈ (kernel read).support,
      observed result = selected read := by
    intro read _supported result reached
    obtain ⟨value, _valueSupported, rfl⟩ := (PMF.mem_support_map_iff _ _ _).mp reached
    rfl
  have observations : actual.map observed = reference.map selected := by
    rw [actualLaw, PMF.map_bind, ← PMF.bind_pure_comp]
    apply bind_congr_on_support reference
    intro read supported
    calc
      (kernel read).map observed = (kernel read).map (fun _ => selected read) :=
        map_congr_on_support _ (fun result reached => determines read supported result reached)
      _ = PMF.pure (selected read) := PMF.map_const _ _
  have referencePossible : accepted ∈ (reference.map selected).support := by
    change accepted ∈ (actual.map observed).support at possible
    rwa [observations] at possible
  change fiberPosterior actual observed accepted = _
  rw [actualLaw, fiberPosterior_bind_of_observation reference kernel selected observed
    determines accepted referencePossible]
  apply bind_congr_on_support (fiberPosterior reference selected accepted)
  intro read reached
  have same := (mem_support_fiberPosterior referencePossible reached).1
  dsimp only [kernel]
  rw [same]

end Vegas
