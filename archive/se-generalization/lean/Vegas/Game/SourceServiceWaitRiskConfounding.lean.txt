/-! # Accepted late binding calls can hide persistent opportunity risk

The real runtime calculation exchanges one clock tick and a fixed semantic
response, including certified openings. The actual submitted packet, post-tick
application state, network and foreign input coincide. The owner's recorded before-view does not:
an unprotected opportunity latches private risk even when its call is accepted.

The separate finite path law retains the common geometric first wait. A fair
scheduler branch selects a protected or an unprotected second owner turn. Both
accepted branches have the same leading waiting weight, so the conditional risk
probability is exactly one half. This is a path calculation, not a certified
whole AsyncServiceSpec, a fully mixed native completion or an SE counterexample.
-/

noncomputable section

namespace Vegas.WaitRiskConfounding

open Interaction EventGraph EventGraphRuntime GameTheory.Math.Probability

variable {Player : Type} [DecidableEq Player]
  {L : IExpr} [IExpr.ResultTypes L] {graph : EventGraph Player L}
  (runtime : EventGraphRuntime graph)
  (leaks : MessageNetwork.ObservationRule Player (WitnessedPacket graph))

local notation "app" => runtime.reactiveApplication leaks

/-- The exact execution produced by the existing clock-advance command. -/
def advance (execution : (app).Execution) : (app).Execution :=
  { execution with
    application := { execution.application with clock := execution.application.clock + 1 }
    environmentRecall := execution.environmentRecall ++
      [⟨execution.observeEnvironment (app), .application .advanceClock⟩] }

theorem advance_law (execution : (app).Execution) :
    execution.environmentStep (app) (.application .advanceClock) =
      PMF.pure (advance runtime leaks execution) := by
  simp [ReactiveApplication.Execution.environmentStep, reactiveApplication,
    environmentStep, advance, PMF.pure_map]

/-- The activation command's genuine endpoint when its passive sample is empty. -/
def activate (execution : (app).Execution) (owner : Player) : (app).Execution :=
  { execution with environmentRecall := execution.environmentRecall ++
      [⟨execution.observeEnvironment (app), .activate owner⟩] }

theorem activate_empty_law (execution : (app).Execution) (owner : Player)
    (empty : leaks owner execution.network.pending = PMF.pure ∅) :
    execution.environmentStep (app) (.activate owner) =
      PMF.pure (activate runtime leaks execution owner) := by
  simp only [ReactiveApplication.Execution.environmentStep, reactiveApplication, empty,
    PMF.pure_map, MessageNetwork.learn_empty, activate]

/-- Packet contents, evidence and readiness tokens are independent of the clock. -/
theorem submission_emit_clock (material : WitnessedSubmission graph)
    (state : EventGraphRuntime.State graph) (owner : Player)
    (known : List (Message Player (WitnessedPacket graph))) (clock : Nat) :
    material.emit { state with clock := clock } owner known =
      material.emit state owner known := by
  rfl

/-- Actual private candidate registration commutes with assigning the clock. -/
theorem submission_register_clock (material : WitnessedSubmission graph)
    (state : EventGraphRuntime.State graph) (owner : Player) (clock : Nat) :
    (app).submit { state with clock := clock } owner material =
      { (app).submit state owner material with clock := clock } := by
  rcases material with ⟨⟨packet, opening⟩, evidence⟩
  cases packet with
  | commitment event candidate =>
    rcases candidate with ⟨actor, slot⟩
    cases slot <;> cases opening <;>
      simp only [reactiveApplication, Submission.register, submitStep]
    all_goals split_ifs <;> rfl
  | opening => rfl
  | withhold => rfl
  | malformed => rfl

/-- Exchanging a tick and the same semantic response preserves the complete
application, network and receipts, including certificate-bearing packets. -/
theorem response_advance_fields (execution : (app).Execution) (owner : Player)
    (response : (app).Action) :
    let early := advance runtime leaks
      ((activate runtime leaks execution owner).respond (app) owner response)
    let late := (activate runtime leaks (advance runtime leaks execution) owner).respond (app)
      owner response
    early.application = late.application ∧ early.network = late.network ∧
      early.receipts = late.receipts := by
  intro early late
  rcases response with ⟨emission⟩
  cases emission with
  | none => exact ⟨rfl, rfl, rfl⟩
  | some material =>
    have clockEq := congrArg PublicView.clock
      (runtime.reactiveApplication_submit_publicView leaks execution.application owner material)
    change ((app).submit execution.application owner material).clock =
      execution.application.clock at clockEq
    refine ⟨?_, ?_, rfl⟩
    · change { (app).submit execution.application owner material with
          clock := ((app).submit execution.application owner material).clock + 1 } =
        (app).submit { execution.application with clock := execution.application.clock + 1 }
          owner material
      rw [clockEq]
      exact (submission_register_clock runtime leaks material execution.application owner _).symm
    · change (execution.network.submit owner
          ((app).packet ((app).submit execution.application owner material) owner
            (execution.network.known owner) material)).2 =
        (execution.network.submit owner
          ((app).packet ((app).submit
            { execution.application with clock := execution.application.clock + 1 } owner material)
            owner (execution.network.known owner) material)).2
      rw [submission_register_clock]
      change _ = (execution.network.submit owner
        (material.emit { (app).submit execution.application owner material with
          clock := execution.application.clock + 1 } owner
            (execution.network.known owner))).2
      rw [submission_emit_clock]
      rfl

/-- After the same actual inclusion, a foreign player has the same full input.
The sender's recorded before-view remains private and may differ. -/
theorem response_advance_foreign_input (execution : (app).Execution) (owner focal : Player)
    (foreign : focal ≠ owner) (response : (app).Action) (id : MessageId Player) :
    let early := (advance runtime leaks
      ((activate runtime leaks execution owner).respond (app) owner response)).includePending
        (app) id
    let late := ((activate runtime leaks (advance runtime leaks execution) owner).respond (app)
      owner response).includePending (app) id
    (early.recall focal, early.observe (app) focal) =
      (late.recall focal, late.observe (app) focal) := by
  intro early late
  obtain ⟨applicationEq, networkEq, receiptsEq⟩ :=
    response_advance_fields runtime leaks execution owner response
  have recalled :
      (advance runtime leaks
        ((activate runtime leaks execution owner).respond (app) owner response)).recall focal =
      ((activate runtime leaks (advance runtime leaks execution) owner).respond (app) owner
        response).recall focal := by
    exact ((app).respond_recall_other (activate runtime leaks execution owner) owner focal foreign
      response).trans
      ((app).respond_recall_other (activate runtime leaks (advance runtime leaks execution) owner)
        owner focal foreign response).symm
  dsimp only [early, late, ReactiveApplication.Execution.includePending]
  rw [networkEq, applicationEq, receiptsEq]
  cases included :
      (((activate runtime leaks (advance runtime leaks execution) owner).respond (app) owner
        response).network.includePending id).1 <;>
    simp only [ReactiveApplication.Execution.observe, recalled]

/-- With deadline three and inclusion bound two, the tick preserves actual
timeliness but turns the unrecorded second owner opportunity into private risk. -/
theorem second_opportunity (bound : graph.EventId → Nat) (execution : (app).Execution)
    (owner : Player) (event : graph.EventId)
    (clock : execution.application.clock = 0)
    (entered : execution.application.activatedAt event = some 0)
    (deadline : runtime.deadline event = 3) (inclusion : bound event = 2)
    (turn : (execution.observe (app) owner).application.publicView.ownTurn? owner = some event)
    (unrecorded : runtime.eventRecorded leaks (execution.recall owner) event = false) :
    runtime.firstUnprotectedOpportunity leaks bound owner (execution.recall owner)
        (execution.observe (app) owner) = false ∧
      runtime.firstUnprotectedOpportunity leaks bound owner
        ((advance runtime leaks execution).recall owner)
        ((advance runtime leaks execution).observe (app) owner) = true ∧
      execution.application.WithinDeadline runtime event ∧
      (advance runtime leaks execution).application.WithinDeadline runtime event := by
  have fits :
      (execution.observe (app) owner).application.publicView.InclusionFitsDeadline runtime bound
        event := by
    change (match execution.application.activatedAt event with
      | none => False
      | some started => execution.application.clock - started + bound event <
          runtime.deadline event)
    rw [entered, clock, inclusion, deadline]
    decide
  have unprotected :
      ¬ PublicView.InclusionFitsDeadline runtime bound
        ((advance runtime leaks execution).observe (app) owner).application.publicView event := by
    change ¬ (match execution.application.activatedAt event with
      | none => False
      | some started => execution.application.clock + 1 - started + bound event <
          runtime.deadline event)
    rw [entered, clock, inclusion, deadline]
    decide
  refine ⟨runtime.firstUnprotectedOpportunity_protected leaks bound owner _ _ event turn fits,
    (runtime.firstUnprotectedOpportunity_iff leaks bound owner _ _).mpr
      ⟨rfl, event, turn, unrecorded, unprotected⟩, ?_, ?_⟩
  · change (match execution.application.activatedAt event with
      | none => False
      | some started => execution.application.clock - started < runtime.deadline event)
    rw [entered, clock, deadline]
    decide
  · change (match execution.application.activatedAt event with
      | none => False
      | some started => execution.application.clock + 1 - started < runtime.deadline event)
    rw [entered, clock, deadline]
    decide

/-- The clock order changes only the owner's private risk record; both
responses transmit the identical bare typed-success commitment. -/
theorem second_binding_risk (bound : graph.EventId → Nat) (execution : (app).Execution)
    (owner : Player) (event : graph.EventId) (payload : L.Ty) (value : L.Val payload) (serial : Nat)
    (clock : execution.application.clock = 0)
    (entered : execution.application.activatedAt event = some 0)
    (deadline : runtime.deadline event = 3) (inclusion : bound event = 2)
    (turn : (execution.observe (app) owner).application.publicView.ownTurn? owner = some event)
    (unrecorded : runtime.eventRecorded leaks (execution.recall owner) event = false)
    (prior : runtime.recalledOpportunityRisk leaks bound owner (execution.recall owner) = false) :
    let response := runtime.reactiveBinding leaks owner event payload (.success value) serial
    runtime.recalledOpportunityRisk leaks bound owner
        (((activate runtime leaks execution owner).respond (app) owner response).recall owner) =
          false ∧
      runtime.recalledOpportunityRisk leaks bound owner
        (((activate runtime leaks (advance runtime leaks execution) owner).respond (app) owner
          response).recall owner) = true := by
  intro response
  obtain ⟨clearOpportunity, unprotected, _earlyTimely, _lateTimely⟩ :=
    second_opportunity runtime leaks bound execution owner event clock entered deadline inclusion
      turn unrecorded
  exact ⟨(runtime.recalledOpportunityRisk_respond_clear leaks bound execution owner response
    clearOpportunity).trans prior,
    runtime.recalledOpportunityRisk_respond_of_opportunity leaks bound
      (advance runtime leaks execution) owner response unprotected⟩

/-- Both actual calls are accepted at the common clock. The conclusion uses
the real receipts and configuration, with no assumed handler/readout equality. -/
theorem second_binding_accepted (execution : (app).Execution) (owner : Player)
    (event : graph.EventId) (payload : L.Ty) (value : L.Val payload)
    (outputEq : graph.outputLayout event = .binding owner payload)
    (codeEq : cast (congrArg (EventCode graph.layout) outputEq)
      (graph.nodes event) = .bind owner payload)
    (node : nodeView graph event = .bind owner payload outputEq codeEq)
    (serial : Nat) (ready : execution.application.config.cut.Ready event)
    (clock : execution.application.clock = 0)
    (entered : execution.application.activatedAt event = some 0)
    (deadline : runtime.deadline event = 3)
    (fresh : execution.application.candidates.lookup (owner, .prepared serial) = .fresh)
    (vacant : execution.application.accepted (.inr event) = none)
    (unused : execution.application.HandleUnused (owner, .prepared serial))
    (serials : execution.network.SerialsBeforeNext) :
    let response := runtime.reactiveBinding leaks owner event payload (.success value) serial
    let id := (owner, execution.network.nextSerial owner)
    let early := (advance runtime leaks
      ((activate runtime leaks execution owner).respond (app) owner response)).includePending
        (app) id
    let late := ((activate runtime leaks (advance runtime leaks execution) owner).respond (app)
      owner response).includePending (app) id
    early.receipts = execution.receipts ++ [(id, true)] ∧
      late.receipts = execution.receipts ++ [(id, true)] ∧
      early.application.config = late.application.config := by
  intro response id early late
  have timely : (advance runtime leaks execution).application.WithinDeadline runtime event := by
    change (match execution.application.activatedAt event with
      | none => False
      | some started => execution.application.clock + 1 - started < runtime.deadline event)
    rw [entered, clock, deadline]
    decide
  let players : Player → (app).Policy := fun _ => (app).silentPolicy
  let network : runtime.NetworkPolicy leaks := fun _ _ => PMF.pure .wait
  have realized := runtime.rawBinding_reserved_config leaks
    (activate runtime leaks (advance runtime leaks execution) owner) owner event payload outputEq
    codeEq node serial (some ⟨payload, value⟩) ready timely fresh vacant unused serials players
      network
  dsimp only at realized
  rw [runtime.rawBinding_reserved_selection leaks
    (activate runtime leaks (advance runtime leaks execution) owner) owner event serial
    (some ⟨payload, value⟩) serials players network] at realized
  simp only [ReactiveApplication.Execution.environmentStep, PMF.pure_map] at realized
  have endpoint := congrArg PMF.support realized
  simp only [PMF.support_pure] at endpoint
  have realizedEndpoint := Set.singleton_injective endpoint
  have received : late.receipts = execution.receipts ++ [(id, true)] :=
    congrArg Prod.snd realizedEndpoint
  obtain ⟨applicationEq, networkEq, receiptsEq⟩ :=
    response_advance_fields runtime leaks execution owner response
  have same : early.application = late.application ∧ early.receipts = late.receipts := by
    dsimp only [early, late, ReactiveApplication.Execution.includePending]
    rw [applicationEq, networkEq, receiptsEq]
    dsimp only [response]
    split <;> exact ⟨rfl, rfl⟩
  exact ⟨same.2.trans received, received, congrArg EventGraphRuntime.State.config same.1⟩

inductive Endpoint where
  | early | clean | privateRisk | missed
  deriving DecidableEq, Fintype

/-- The selected second-turn branch keeps the actual common geometric
weight `(1-weight)*weight`; the scheduler coin is independent of private data. -/
def pathLaw (weight : ℝ) (nonnegative : 0 ≤ weight) (bounded : weight ≤ 1) : PMF Endpoint :=
  (geometricTurnLaw weight nonnegative bounded 2).bind fun slot =>
    if slot.val = 0 then PMF.pure .early
    else if slot.val = 1 then mix (1 / 2) (by norm_num) (by norm_num)
      (PMF.pure .privateRisk) (PMF.pure .clean)
    else PMF.pure .missed

theorem pathLaw_late_mass (weight : ℝ) (nonnegative : 0 ≤ weight) (bounded : weight ≤ 1) :
    ((pathLaw weight nonnegative bounded) .clean).toReal = (1 - weight) * weight / 2 ∧
      ((pathLaw weight nonnegative bounded) .privateRisk).toReal =
        (1 - weight) * weight / 2 := by
  unfold pathLaw
  simp only [toReal_bind_apply, expect_eq_sum, Fin.sum_univ_succ]
  simp only [Fin.val_zero, ↓reduceIte, PMF.pure_apply, reduceCtorEq, ENNReal.toReal_zero,
    mul_zero, Fin.val_succ, Nat.reduceAdd, mix_apply_toReal,
    ENNReal.toReal_one, mul_one, zero_add, add_zero]
  rw [geometricTurnLaw_apply_toReal]
  norm_num
  ring

/-- The common foreign wait cancels from the conditional ratio. No choice
of a positive event-only geometric weight can make this conditional mass small. -/
theorem pathLaw_private_given_late (weight : ℝ) (positive : 0 < weight) (small : weight < 1) :
    ((pathLaw weight positive.le small.le) .privateRisk).toReal /
        (((pathLaw weight positive.le small.le) .clean).toReal +
          ((pathLaw weight positive.le small.le) .privateRisk).toReal) = 1 / 2 := by
  obtain ⟨clean, risky⟩ := pathLaw_late_mass weight positive.le small.le
  rw [clean, risky]
  have nonzero : (1 - weight) * weight ≠ 0 :=
    (mul_pos (sub_pos.mpr small) positive).ne'
  apply (div_eq_iff (by positivity)).mpr
  ring
