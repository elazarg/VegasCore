import Vegas.Game.SourceServicePublicRejection

/-! # New packets addressed to completed events

An old accepted envelope remains permitted. A new identifier allocated after
the named event completed cannot acquire an accepting receipt, even when its
readiness token and signed content are valid. The proof follows actual emitted
envelopes and arbitrary later responses and scheduler commands.
-/

noncomputable section

namespace Vegas

open SourceProgram Interaction EventGraphRuntime GameTheory.Protocol GameTheory.Math.Probability

variable {Player : Type} [DecidableEq Player]
  {L : IExpr} [IExpr.ResultTypes L]
  {setup : Setup (Player := Player) (L := L)}
  {leaks : MessageNetwork.ObservationRule Player (WitnessedPacket (graph setup))}

/-- A new packet allocated after its addressed event completed cannot acquire
an accepting receipt under arbitrary later responses and scheduler commands.
The fresh identifier premise distinguishes it from the original accepted packet. -/
theorem completedPacket_not_accepted_reaches
    {horizon : Nat} {scheduler : (application setup leaks).Scheduler}
    {first last :
      ((application setup leaks).protocol (initialLaw setup) horizon scheduler).History}
    {fuel : Nat}
    (path : ((application setup leaks).protocol (initialLaw setup) horizon scheduler).ReachesWithin
      fuel first last)
    (before after : (application setup leaks).Control)
    (firstState : first.state = some before) (lastState : last.state = some after)
    (event : (graph setup).EventId)
    (completed : event ∈ before.execution.application.config.cut.completed)
    (message : Message Player (WitnessedPacket (graph setup)))
    (named : message.payload.call.event? (graph setup) = some event)
    (fresh : message.id = (message.sender, before.execution.network.nextSerial message.sender))
    (emitted : Emitted setup leaks after.execution message) :
    (message.id, true) ∉ after.execution.receipts := by
  exact publiclyBlockedPacket_not_accepted_reaches path before after firstState lastState event
    message (Or.inl completed) named fresh emitted

/-- The actual final settled record forbids this newly emitted completed-event
packet. Valid persistent readiness tokens and lawful signed content do not alter
the absence of an accepting receipt. -/
theorem completedPacket_forbidden_reaches
    {horizon : Nat} {scheduler : (application setup leaks).Scheduler}
    {first last :
      ((application setup leaks).protocol (initialLaw setup) horizon scheduler).History}
    {fuel : Nat}
    (path : ((application setup leaks).protocol (initialLaw setup) horizon scheduler).ReachesWithin
      fuel first last)
    (before after : (application setup leaks).Control)
    (firstState : first.state = some before) (lastState : last.state = some after)
    (event : (graph setup).EventId)
    (completed : event ∈ before.execution.application.config.cut.completed)
    (message : Message Player (WitnessedPacket (graph setup)))
    (named : message.payload.call.event? (graph setup) = some event)
    (fresh : message.id = (message.sender, before.execution.network.nextSerial message.sender))
    (emitted : Emitted setup leaks after.execution message)
    (complete : after.execution.application.config.cut.Terminal) :
    ((runtime setup).settledRecord leaks after.execution).permits message = false := by
  exact publiclyBlockedPacket_forbidden_reaches path before after firstState lastState event
    message (Or.inl completed) named fresh emitted complete

end Vegas
