import Vegas.Game.SourceServiceUnusableBinding
import Vegas.Game.SourceServiceProtectedBinding

/-! # The actual protected call of an unusable binding residual

A clear legal risk-menu input supplies a fresh counted candidate and protected
inclusion time. The excluded binding's absent or mistyped private material does
not enter its bare public packet or the handler's public acceptance conditions.
The actual response therefore appends an acceptable protected fresh call.

At a completed raw continuation, protected inclusion records the selected handle
provided its actual identifier remains the owner's sole identifier for this
event. This condition stays explicit: a later raw retry can void protection.
No source strategy, private value admission or evidence-collection hypothesis is
used, and no continuation payoff comparison is asserted.
-/

noncomputable section

namespace Vegas

open SourceProgram Interaction EventGraphRuntime EventGraph

variable {Player : Type} [DecidableEq Player] [Fintype Player]
  {L : IExpr} [IExpr.ResultTypes L]
  {setup : Setup (Player := Player) (L := L)}
  {leaks : MessageNetwork.ObservationRule Player (WitnessedPacket (graph setup))}

/-- The residual response's real emitted packet and recalled entry form a
protected fresh call, regardless of whether its private material is absent or
mistyped. Naming the event is only the response's syntactic submitted-event fact. -/
theorem unusableServiceBindingResponse_protected_call
    (bounds : MessageBounds (graph setup)) (bound : (graph setup).EventId → Nat)
    {horizon remaining : Nat} {scheduler : (application setup leaks).Scheduler}
    (execution : (application setup leaks).Execution) (who : Player)
    (trace : ((bounds.riskMenu (runtime setup) leaks bound).protocol (initialLaw setup) horizon
      scheduler).Trace (some ⟨remaining, some who, execution⟩))
    (clear : (runtime setup).serviceRisk leaks bound who (execution.recall who)
      (execution.observe (application setup leaks) who) = false)
    (response : (application setup leaks).Action)
    (unusable : unusableServiceBindingResponse setup leaks who (execution.recall who)
      (execution.observe (application setup leaks) who) response)
    (event : (graph setup).EventId)
    (named : (runtime setup).submittedEvent? leaks response = some event) :
    ∃ (payload : L.Ty) (opening : Option (Raw L)),
      let app := application setup leaks
      let candidate : Handle (graph setup) :=
        (who, .prepared (execution.application.publicView.bindingCount who))
      let material : app.Submission := ⟨⟨.commitment event candidate, opening⟩, .none⟩
      let message : Message Player (WitnessedPacket (graph setup)) :=
        ⟨(who, execution.network.nextSerial who),
          ⟨.commitment event candidate, none, some ⟨event⟩⟩⟩
      let entry : app.PlayerEntry := ⟨execution.observe app who, response, some message⟩
      (graph setup).actor? event = some who ∧
        (graph setup).outputLayout event = .binding who payload ∧
        execution.application.config.cut.Ready event ∧
        (runtime setup).eventRecorded leaks (execution.recall who) event = false ∧
        execution.application.candidates.lookup candidate = .fresh ∧
        execution.application.publicView.InclusionFitsDeadline (runtime setup) bound event ∧
        response = ⟨some material⟩ ∧
        opening.bind (fun raw => raw.as? payload) = none ∧
        app.packet (app.submit execution.application who material) who
          (execution.network.known who) material = message.payload ∧
        (execution.respond app who response).recall who = execution.recall who ++ [entry] ∧
        FreshCall setup leaks who event bound entry message := by
  obtain ⟨actual, payload, outputEq, codeEq, node, turn, unrecorded, opening, responseEq,
    missing⟩ := unusable
  have addressed : (runtime setup).submittedEvent? leaks response = some actual := by
    rw [responseEq]
    rfl
  have same : actual = event := Option.some.inj (addressed.symm.trans named)
  subst actual
  let app := application setup leaks
  let candidate : Handle (graph setup) :=
    (who, .prepared (execution.application.publicView.bindingCount who))
  let material : app.Submission := ⟨⟨.commitment event candidate, opening⟩, .none⟩
  let message : Message Player (WitnessedPacket (graph setup)) :=
    ⟨(who, execution.network.nextSerial who),
      ⟨.commitment event candidate, none, some ⟨event⟩⟩⟩
  let entry : app.PlayerEntry := ⟨execution.observe app who, response, some message⟩
  have readyView := (PublicView.ownTurn?_spec execution.application.publicView who event turn).1
  have ready := (execution.application.publicView_eventReady event).mp readyView
  have owned := nodeView_bind_actor outputEq codeEq
  have fits := (runtime setup).serviceRisk_clear_protected_opportunity leaks bound who
    (execution.recall who) (execution.observe app who) event rfl turn unrecorded clear
  have persistent := ((runtime setup).serviceRisk_clear_iff leaks bound who _ _).mp clear |>.1
  have fresh := riskCanonicalSlot_fresh_at_turn bounds bound _ trace who persistent event turn
    unrecorded
  have rawTrace := (bounds.riskMenu (runtime setup) leaks bound).toRawTrace (initialLaw setup)
    horizon scheduler trace
  have facts := legalFacts setup leaks horizon scheduler _ rawTrace
  have vacant : execution.application.accepted (.inr event) = none := by
    cases associated : execution.application.accepted (.inr event) with
    | none => rfl
    | some selected =>
        exact False.elim (ready.1
          (facts.binding.toAssociationInvariant.accepted_complete event selected associated))
  have unused : execution.application.HandleUnused candidate :=
    fun field associated => facts.binding.accepted_fixed field candidate associated fresh
  have packet : app.packet (app.submit execution.application who material) who
      (execution.network.known who) material = message.payload := by
    rw [reactiveApplication_packet_none,
      execution.application.publicView_tokenFor_of_ready _ event rfl ready]
  have conforms : (runtime setup).freshServiceEnvelope execution.application.publicView message :=
    ((runtime setup).freshServiceEnvelope_binding_iff execution.application.publicView
      message.id event candidate none (some ⟨event⟩)).mpr
      ⟨by
        simp only [PublicView.BindingIncludable, node]
        exact ⟨readyView, fits.withinDeadline, rfl, rfl, vacant, unused⟩,
        rfl, rfl, rfl⟩
  have responseShape : response = ⟨some material⟩ := responseEq
  have recalled : (execution.respond app who response).recall who =
      execution.recall who ++ [entry] := by
    change (execution.respond app who response).recall who =
      execution.recall who ++ [⟨execution.observe app who, response, some message⟩]
    rw [responseShape]
    simp only [ReactiveApplication.Execution.respond, ↓reduceIte]
    rw [packet]
    rfl
  refine ⟨payload, opening, owned, outputEq, ready, unrecorded, fresh, fits, responseEq,
    missing, packet, recalled, ?_⟩
  exact {
    fresh := ⟨material, congrArg ReactiveApplication.Action.transmission responseEq⟩
    emitted := rfl
    authored := rfl
    addressed := rfl
    ready := readyView
    fits := fits
    conforming := EventGraphRuntime.freshServiceEnvelope.acceptable (runtime setup) conforms }

/-- When the actual residual call remains the sole owner identifier until its
event completes, its counted handle is used in the actual contract record.
Foreign policies and later raw responses are unrestricted subject to this
explicit sole-identifier condition. -/
theorem unusableServiceBindingResponse_completed_association
    (bounds : MessageBounds (graph setup)) (bound : (graph setup).EventId → Nat)
    {horizon remaining : Nat} {scheduler : (application setup leaks).Scheduler}
    (inclusion : ProtectedInclusion (runtime setup) leaks (initialLaw setup) horizon scheduler
      bound)
    (execution : (application setup leaks).Execution) (who : Player)
    (trace : ((bounds.riskMenu (runtime setup) leaks bound).protocol (initialLaw setup) horizon
      scheduler).Trace (some ⟨remaining, some who, execution⟩))
    (clear : (runtime setup).serviceRisk leaks bound who (execution.recall who)
      (execution.observe (application setup leaks) who) = false)
    (response : (application setup leaks).Action)
    (unusable : unusableServiceBindingResponse setup leaks who (execution.recall who)
      (execution.observe (application setup leaks) who) response)
    (event : (graph setup).EventId)
    (named : (runtime setup).submittedEvent? leaks response = some event)
    (control : (application setup leaks).Control)
    (finalTrace : ((application setup leaks).protocol (initialLaw setup) horizon scheduler).Trace
      (some control))
    (later : List (application setup leaks).PlayerEntry)
    (continued : control.execution.recall who =
      (execution.respond (application setup leaks) who response).recall who ++ later)
    (sole : ∀ other ∈ execution.recall who ++ later,
      ¬ EmitsOtherFor (runtime setup) leaks other event (who, execution.network.nextSerial who))
    (completed : event ∈ control.execution.application.config.cut.completed) :
    ((who, execution.network.nextSerial who), true) ∈ control.execution.receipts ∧
      control.execution.application.publicView.accepted (.inr event) =
        some (who, .prepared (execution.application.publicView.bindingCount who)) ∧
      event ∉ control.execution.application.missedEvents := by
  obtain ⟨payload, opening, owned, _output, _ready, _unrecorded, _fresh, _fits, _response,
    _missing, _packet, recalled, call⟩ :=
    unusableServiceBindingResponse_protected_call bounds bound execution who trace clear response
      unusable event named
  let app := application setup leaks
  let candidate : Handle (graph setup) :=
    (who, .prepared (execution.application.publicView.bindingCount who))
  let message : Message Player (WitnessedPacket (graph setup)) :=
    ⟨(who, execution.network.nextSerial who),
      ⟨.commitment event candidate, none, some ⟨event⟩⟩⟩
  let entry : app.PlayerEntry := ⟨execution.observe app who, response, some message⟩
  have split : control.execution.recall who = execution.recall who ++ entry :: later := by
    rw [continued, recalled, List.append_assoc]
    rfl
  have accepted := (protected_binding_no_miss setup leaks inclusion finalTrace event who candidate
    owned (execution.recall who) later entry message split call rfl sole).1 completed
  have settled := settlesFreshCalls_history setup leaks inclusion who event owned finalTrace
    (execution.recall who) entry later message split call sole
  exact ⟨settled.1 completed, accepted, settled.2.2.2⟩

end Vegas
