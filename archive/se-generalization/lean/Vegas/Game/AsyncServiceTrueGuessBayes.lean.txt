/-! # The actual conditional HIGH comparator

A compatible information input supplies one successful TRUE action. Genuine
normalized source support admits its canonical packet. The same implementable
whole comparator obtains the actual conditional HIGH-type probability, keeping
all foreign waiting and free-menu likelihoods in its counterfactual weights.
-/

noncomputable section

/-- Actual source residual decoding and normalized support suffice for the
canonical TRUE packet at any compatible current resolution. -/
theorem sourceCompatibleInfo_true_supported_of_node
    (profile : BehavioralProfile service.setup.program)
    (supports : ∀ who, (profile who).SupportsEffectiveChoices service.setup.program
      (CommitmentInterface.values service.setup.program) []
      (Revelations.initial service.setup.context))
    (remaining : Nat) (execution : (app).Execution) (who : Player)
    (trace : ((menu).protocol (initialLaw service.setup) service.horizon service.scheduler).Trace
      (some ⟨remaining, some who, execution⟩))
    (compatible : service.sourceCompatibleInfo who
      (some (execution.recall who, execution.observe (app) who)))
    (event : (graph service.setup).EventId) (payload : L.Ty)
    (binding : FieldRef (graph service.setup).layout (.binding who payload))
    (checks : List (GuardCheck (graph service.setup).layout payload))
    (outputEq : (graph service.setup).outputLayout event = .publication payload)
    (codeEq : cast (congrArg (EventCode (graph service.setup).layout) outputEq)
      ((graph service.setup).nodes event) = .resolve who payload binding checks)
    (turn : execution.application.publicView.ownTurn? who = some event)
    (unrecorded : (runtime).eventRecorded service.leaks (execution.recall who) event = false)
    (value : L.Val payload)
    (successful : EventCode.resolveOutput? binding checks true
      execution.application.config.store = some (.success value)) :
    (runtime).canonicalServiceDecision service.leaks who (execution.recall who)
      (execution.observe (app) who) event
        (cast (congrArg EventField.Action outputEq.symm) true) ∈
      (sourceServiceImmediatePolicy service.setup service.leaks service.bound profile who
        (execution.recall who) (execution.observe (app) who)).support := by
  have ready := (execution.application.publicView_eventReady event).mp
    (PublicView.ownTurn?_spec _ who event turn).1
  have owned := (PublicView.ownTurn?_spec _ who event turn).2
  obtain ⟨residual⟩ := menu_ready_sourceResidual service.setup service.leaks profile (menu)
    service.horizon service.scheduler trace ⟨remaining, some who, execution⟩ rfl event ready
  obtain ⟨site⟩ := residual.revealSource outputEq
  change RevealSource service.setup profile event execution.application.config at site
  have sameOwner : site.owner = who := Option.some.inj (site.owned.symm.trans owned)
  have samePayload : site.payload = payload :=
    EventField.publication.inj (site.outputEq.symm.trans outputEq)
  subst who payload
  have sameCode := codeEq.symm.trans site.code
  have same : binding = site.refs.get site.binding ∧ checks =
      compileChecks (published := site.published) site.refs site.source.registry
        site.source.revelations site.binding := by
    simpa using (EventCode.resolve.inj sameCode)
  obtain ⟨rfl, rfl⟩ := same
  exact service.sourceCompatibleInfo_immediate_true_supported profile supports remaining
    execution event site ((menu).toRawTrace (initialLaw service.setup) service.horizon
      service.scheduler trace) compatible turn unrecorded value successful

/-- One actual local TRUE packet followed by the same whole immediate policy
has the native HIGH-type Bayes value. No source posterior is substituted. -/
theorem bayes_true_guess_comparator_eq
    (parameter : State L service.setup.context → Bool)
    (source : BehavioralProfile service.setup.program) (who : Player)
    (permitted : (source who).Admitted service.setup.program (CommitmentInterface.values _))
    (supports : ∀ player, (source player).SupportsEffectiveChoices service.setup.program
      (CommitmentInterface.values service.setup.program) []
      (Revelations.initial service.setup.context))
    (native : ∀ player, (nativeModel).BehavioralPolicy player)
    (site : (nativeModel).InformationSite who)
    (positive : 0 < (nativeModel).informationMass native who site)
    (past : List (app).PlayerEntry) (view : (app).PlayerView)
    (observed : site.1 = some (past, view))
    (compatible : service.sourceCompatibleInfo who site.1)
    (event : (graph service.setup).EventId) (payload : L.Ty)
    (binding : FieldRef (graph service.setup).layout (.binding who payload))
    (checks : List (GuardCheck (graph service.setup).layout payload))
    (outputEq : (graph service.setup).outputLayout event = .publication payload)
    (codeEq : cast (congrArg (EventCode (graph service.setup).layout) outputEq)
      ((graph service.setup).nodes event) = .resolve who payload binding checks)
    (turn : view.application.publicView.ownTurn? who = some event)
    (unrecorded : (runtime).eventRecorded service.leaks past event = false)
    (value : L.Val payload)
    (successful : EventCode.resolveOutput? binding checks true view.application.observation.store =
      some (.success value))
    (choice : (nativeModel).Choice who site.1)
    (selected : choice.1 = some ((runtime).canonicalServiceDecision service.leaks who past view
      event (cast (congrArg EventField.Action outputEq.symm) true)))
    (sample : List (SettledEvidence service.setup) → PMF (List (SettledEvidence service.setup)))
    (authentic : ∀ actual sampled, sampled ∈ (sample actual).support → sampled ⊆ actual)
    (deposit : Player → ℝ) :
    let belief := (nativeModel).bayesBelief native who site
      ((menu).decisionInformationAntichain (initialLaw service.setup) service.horizon
        service.scheduler who site) positive
    let alternative := (service.effectiveImmediateComparator source who).commit site.1 choice
    expect belief (fun history =>
      expect ((nativeModel).runBehavioralTerminalFrom
        ((menu).bounded (initialLaw service.setup) service.horizon
          service.scheduler).wellFoundedHistories
        (Profile.update (sig := (nativeModel).behavioralSignature) native who alternative)
          history.1)
        (fun final => TerminalAudit.utility
          (fun state _ => sourcePublicationGuessValue service.setup service.leaks event payload
            outputEq (service.initialGuessParameter (menu) parameter history.1) state)
          ((runtime).serviceAuditObservation service.leaks)
          (sourceServiceAudit service.setup service.leaks sample) deposit final.state who)) =
      service.counterfactualInformationMass (menu) native who site
        {history | service.initialGuessParameter (menu) parameter history.1 = true} /
      service.counterfactualInformationMass (menu) native who site Set.univ := by
  intro belief alternative
  let high : Set ((nativeModel).InformationHistory who site.1) :=
    {history | service.initialGuessParameter (menu) parameter history.1 = true}
  have exactValue (history : (nativeModel).InformationHistory who site.1) :
      expect ((nativeModel).runBehavioralTerminalFrom
        ((menu).bounded (initialLaw service.setup) service.horizon
          service.scheduler).wellFoundedHistories
        (Profile.update (sig := (nativeModel).behavioralSignature) native who alternative)
          history.1)
        (fun final => TerminalAudit.utility
          (fun state _ => sourcePublicationGuessValue service.setup service.leaks event payload
            outputEq (service.initialGuessParameter (menu) parameter history.1) state)
          ((runtime).serviceAuditObservation service.leaks)
          (sourceServiceAudit service.setup service.leaks sample) deposit final.state who) =
        high.indicator (fun _ => (1 : ℝ)) history := by
    have active := InformationModel.InformationSite.active (nativeModel) site history
    have input := ((menu).info (initialLaw service.setup) service.horizon service.scheduler who
      history.1.trace).symm.trans (history.2.trans observed)
    cases current : history.1.state with
    | none => rw [current] at active; cases active
    | some control =>
        rcases control with ⟨remaining, actor, execution⟩
        rw [current] at active input
        change actor = some who at active
        subst actor
        have pair : (execution.recall who, execution.observe (app) who) = (past, view) :=
          Option.some.inj (by simpa only [ReactiveApplication.observe, ↓reduceIte] using input)
        have recalled : execution.recall who = past := congrArg Prod.fst pair
        have viewed : execution.observe (app) who = view := congrArg Prod.snd pair
        have actualCompatible : service.sourceCompatibleInfo who
            (some (execution.recall who, execution.observe (app) who)) := by
          rw [pair, ← observed]
          exact compatible
        have actualTurn : execution.application.publicView.ownTurn? who = some event :=
          (congrArg (fun observed : (app).PlayerView =>
            observed.application.publicView.ownTurn? who) viewed).trans turn
        have actualUnrecorded : (runtime).eventRecorded service.leaks
            (execution.recall who) event = false :=
          (congrArg (fun entries => (runtime).eventRecorded service.leaks entries event)
            recalled).trans unrecorded
        have actualSuccess : EventCode.resolveOutput? binding checks true
            execution.application.config.store = some (.success value) := by
          have own : EventCode.resolveOutput? binding checks true
              ((graph service.setup).playerStore who execution.application.config.store) =
              some (.success value) :=
            (congrArg (fun observed : (app).PlayerView =>
              EventCode.resolveOutput? binding checks true observed.application.observation.store)
                viewed).trans successful
          simpa only [EventCode.resolveOutput?_playerStore] using own
        have chosen := service.sourceCompatibleInfo_true_supported_of_node source supports
          remaining execution who (current ▸ history.1.trace) actualCompatible event payload
          binding checks outputEq codeEq actualTurn actualUnrecorded value actualSuccess
        let localChoice : (nativeModel).Choice who ((nativeModel).infoOf who history.1.trace) :=
          history.2.symm ▸ choice
        have transport (info other : (nativeModel).InfoState who) (same : info = other)
            (action : (nativeModel).Choice who other) :
            let localAction : (nativeModel).Choice who info := same.symm ▸ action
            localAction.1 = action.1 ∧
              (service.effectiveImmediateComparator source who).commit info localAction =
                (service.effectiveImmediateComparator source who).commit other action := by
          cases same
          exact ⟨rfl, rfl⟩
        obtain ⟨valueEq, policyEq⟩ := transport _ _ history.2 choice
        have localSelected : localChoice.1 = some ((runtime).canonicalServiceDecision
            service.leaks who (execution.recall who) (execution.observe (app) who) event
              (cast (congrArg EventField.Action outputEq.symm) true)) := by
          rw [show localChoice.1 = choice.1 from valueEq, recalled, viewed]
          exact selected
        have actual := service.effectiveImmediateComparator_true_guess_value source who permitted
          native history.1 remaining execution current actualCompatible event payload binding
          checks outputEq codeEq (nodeView_eq_resolve outputEq codeEq) actualTurn actualUnrecorded
          value actualSuccess localChoice localSelected chosen sample authentic
          (service.initialGuessParameter (menu) parameter history.1) deposit
        rw [policyEq] at actual
        refine actual.trans ?_
        cases parameterEq : service.initialGuessParameter (menu) parameter history.1 <;>
          simp [high, parameterEq]
  calc
    _ = expect belief (high.indicator (fun _ => (1 : ℝ))) := expect_congr_on_support
      (fun history _ => exactValue history)
    _ = (belief.toOuterMeasure high).toReal := expect_indicator belief high
    _ = _ := service.bayesBelief_event_counterfactual (menu) native who site positive high
