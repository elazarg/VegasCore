/-! # Source residuals at every legal native ready site

Every legal menu history has an aligned source residual at its actual completed
prefix, including histories created by arbitrary deviations. Responses keep
the graph configuration, and scheduler transitions preserve residuals through
the actual graph step. A ready event identifies the residual rank without a
calendar position or a prescribed source-policy support assumption.

The decoder retains the effective source configuration. Restoring original
failed disclosure intentions and transporting equilibrium beliefs are separate.
-/

noncomputable section

namespace Vegas

open SourceProgram Interaction EventGraphRuntime GameTheory.Math.Probability

variable {Player : Type} [DecidableEq Player]
  {L : IExpr} [IExpr.ResultTypes L]
  (setup : Setup (Player := Player) (L := L))
  (leaks : MessageNetwork.ObservationRule Player (WitnessedPacket (graph setup)))
  (profile : BehavioralProfile setup.program)

/-- An aligned source residual exists along every actual legal menu trace,
without a source-strategy admission or full-support premise. -/
theorem menu_history_sourceResidual
    (menu : (application setup leaks).ResponseMenu) (horizon : Nat)
    (scheduler : (application setup leaks).Scheduler)
    {state : (application setup leaks).ProtocolState}
    (trace : (menu.protocol (initialLaw setup) horizon scheduler).Trace state)
    (control : (application setup leaks).Control) (current : state = some control) :
    ∃ rank, Nonempty (SourceResidual setup profile rank control.execution.application.config) := by
  let predicate := fun execution : (application setup leaks).Execution =>
    ∃ rank, Nonempty (SourceResidual setup profile rank execution.application.config)
  have preserved : menu.ServiceInvariant scheduler predicate := {
    respond := by
      rintro execution who response ⟨rank, residual⟩ _member
      refine ⟨rank, ?_⟩
      rw [((runtime setup).reactive_respond_application leaks execution who response).1]
      exact residual
    environment := by
      rintro execution next command ⟨rank, ⟨residual⟩⟩ _selected moved
      have step := environmentStep_configStep setup leaks execution next command moved
      rcases step.prefix rank residual.checkpoint.ordered with same | advanced
      · refine ⟨rank, residual.configStep step rank ?_⟩
        rw [same]
        exact residual.checkpoint.ordered
      · exact ⟨rank + 1, residual.configStep step (rank + 1) advanced⟩ }
  have initialized : ∀ state ∈ (initialLaw setup).support,
      predicate (ReactiveApplication.Execution.initial (application setup leaks) state) := by
    intro state supported
    obtain ⟨initial, _supported, rfl⟩ := PMF.support_map .. ▸ supported
    exact ⟨0, ⟨SourceResidual.initial setup profile initial⟩⟩
  have reached := preserved.history (initialLaw setup) horizon initialized trace
  rw [current] at reached
  exact reached

/-- Actual readiness fixes the source residual rank at the event identifier,
for every legal native history and every menu. -/
theorem menu_ready_sourceResidual
    (menu : (application setup leaks).ResponseMenu) (horizon : Nat)
    (scheduler : (application setup leaks).Scheduler)
    {state : (application setup leaks).ProtocolState}
    (trace : (menu.protocol (initialLaw setup) horizon scheduler).Trace state)
    (control : (application setup leaks).Control) (current : state = some control)
    (event : (graph setup).EventId)
    (ready : control.execution.application.config.cut.Ready event) :
    Nonempty (SourceResidual setup profile event.val control.execution.application.config) := by
  obtain ⟨rank, ⟨residual⟩⟩ :=
    menu_history_sourceResidual setup leaks profile menu horizon scheduler trace control current
  have same := (ready_iff_rank setup control.execution.application.config rank
    residual.checkpoint.ordered event).mp ready
  rw [same]
  exact ⟨residual⟩

variable {setup profile}

/-- A source residual at a binding event is an aligned source commitment,
including when arbitrary earlier native deviations created the residual. -/
theorem SourceResidual.bindingSource {event : (graph setup).EventId}
    {config : (graph setup).Config} (residual : SourceResidual setup profile event.val config)
    {owner : Player} {payload : L.Ty}
    (isBinding : (graph setup).outputLayout event = .binding owner payload) :
    Nonempty (BindingSource setup profile event config) := by
  obtain ⟨Γ, names, program, residualProfile, source, refs, embedding, refsBefore, aligned,
    _admitted, _effective, _supports, _lift, _liftView, _observeLift, _recoverView, _viewRecovered,
    _commutes, _steps,
    _injective, _transport,
    checkpoint⟩ := residual
  cases program with
  | ret result =>
      have count := aligned.graphSuffix.countEq
      simp only [eventCount, Nat.add_zero] at count
      have inside := event.isLt
      change event.val < eventCount setup.program at inside
      omega
  | sample name fresh law next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      have actor := aligned.actorEq ⟨0, by simp [eventCount]⟩
      rw [head] at actor
      change (graph setup).actor? event = none at actor
      rw [binding_actor setup event owner payload isBinding] at actor
      cases actor
  | @commit Γ names name owner payload fresh guard next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      exact ⟨⟨Γ, names, name, owner, payload, fresh, guard, next, residualProfile, refs, source,
        embedding, refsBefore, aligned, checkpoint.agrees, checkpoint.history, head⟩⟩
  | @reveal Γ names published owner name payload fresh binding unresolved next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      have output : (graph setup).outputLayout event = .publication payload := by
        rw [← head]
        change outputLayout setup.program (embedding.event _) = _
        simpa [outputLayout, eventCount] using embedding.layout_eq ⟨0, by simp [eventCount]⟩
      cases output.symm.trans isBinding

/-- A source residual at a publication event is an aligned source disclosure.
Source effectiveness and supported choices are inherited as implications,
rather than assumed of the native history. -/
theorem SourceResidual.revealSource {event : (graph setup).EventId}
    {config : (graph setup).Config} (residual : SourceResidual setup profile event.val config)
    {payload : L.Ty}
    (isPublication : (graph setup).outputLayout event = .publication payload) :
    Nonempty (RevealSource setup profile event config) := by
  obtain ⟨Γ, names, program, residualProfile, source, refs, embedding, refsBefore, aligned,
    _admitted, effective, supports, _lift, _liftView, _observeLift, _recoverView, _viewRecovered,
    _commutes, _steps,
    _injective, _transport,
    checkpoint⟩ := residual
  cases program with
  | ret result =>
      have count := aligned.graphSuffix.countEq
      simp only [eventCount, Nat.add_zero] at count
      have inside := event.isLt
      change event.val < eventCount setup.program at inside
      omega
  | sample name fresh law next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      have layout := embedding.layout_eq ⟨0, by simp [eventCount]⟩
      simp only [outputLayout, eventCount] at layout
      rw [head] at layout
      cases layout.symm.trans isPublication
  | @commit Γ names name owner payload fresh guard next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      have output : (graph setup).outputLayout event = .binding owner payload := by
        rw [← head]
        change outputLayout setup.program (embedding.event _) = _
        simpa [outputLayout, eventCount] using embedding.layout_eq ⟨0, by simp [eventCount]⟩
      cases output.symm.trans isPublication
  | @reveal Γ names published owner name payload fresh binding unresolved next =>
      have head : embedding.event ⟨0, by simp [eventCount]⟩ = event := by
        apply Fin.ext
        simpa only [Nat.add_zero] using aligned.graphSuffix.rankEq ⟨0, by simp [eventCount]⟩
      exact ⟨⟨Γ, names, published, owner, name, payload, fresh, binding, unresolved, next,
        residualProfile, refs, source, embedding, refsBefore, aligned, checkpoint.agrees,
        checkpoint.history, head, effective, supports⟩⟩
