/-! # Actual conditional resolution draws and their distinct traffic

The same native Bayes histories supply an aligned effective source resolution.
Supported effective FALSE and TRUE choices retain their own canonical packet
and full post-response traffic. The native uniform branch and every foreign
waiting or free excursion remain in the exact conditional law. Decoded source
histories are effective histories; failed TRUE intentions are not restored by
this readout.
-/

noncomputable section

private theorem canonical_resolution_joint
    (profile : BehavioralProfile service.setup.program)
    (effective : ∀ player, (profile player).EffectiveDisclosures service.setup.program []
      (Revelations.initial service.setup.context))
    (event : (graph service.setup).EventId) (execution : (app).Execution)
    (source : RevealSource service.setup profile event execution.application.config)
    (remaining : Nat)
    (trace : ((menu).protocol (initialLaw service.setup) service.horizon service.scheduler).Trace
      (some ⟨remaining, some source.owner, execution⟩))
    (ready : execution.application.config.cut.Ready event) :
    (sourceServiceCanonicalPolicy service.setup service.leaks profile source.owner
      (execution.recall source.owner) (execution.observe (app) source.owner)).map
        (fun response => (resolutionResponseDecision? service.setup service.leaks event response,
          execution.respond (app) source.owner response)) =
      (revealKernel source.residual (source.source.view source.owner)).map fun disclose =>
        (some disclose, execution.respond (app) source.owner
          ((runtime).canonicalServiceDecision service.leaks source.owner
            (execution.recall source.owner) (execution.observe (app) source.owner) event
            (cast (congrArg EventGraph.EventField.Action source.outputEq.symm) disclose))) := by
  rw [RevealSource.canonical_response_law execution source ready, PMF.map_comp]
  apply map_congr_on_support
  intro disclose supported
  apply Prod.ext
  · exact RevealSource.response_decision execution
      ((menu).toRawTrace (initialLaw service.setup) service.horizon service.scheduler trace)
        source effective disclose supported
  · rfl

private theorem resolution_pin_joint
    (profile : BehavioralProfile service.setup.program) (who : Player)
    (permitted : (profile who).Admitted service.setup.program (CommitmentInterface.values _))
    (event : (graph service.setup).EventId) (execution : (app).Execution) (remaining : Nat)
    (trace : ((menu).protocol (initialLaw service.setup) service.horizon service.scheduler).Trace
      (some ⟨remaining, some who, execution⟩))
    (compatible : service.sourceCompatibleInfo who
      (some (execution.recall who, execution.observe (app) who)))
    (turn : execution.application.publicView.ownTurn? who = some event)
    (unrecorded : (runtime).eventRecorded service.leaks (execution.recall who) event = false)
    (weight : Player → (app).Info → ℝ)
    (nonnegative : ∀ player info, 0 ≤ weight player info)
    (small : ∀ player info, weight player info ≤ 1)
    (delta : ℝ) (deltaNonnegative : 0 ≤ delta) (deltaSmall : delta ≤ 1)
    (continuation : ∀ player, (model).BehavioralPolicy player) :
    let info := some (execution.recall who, execution.observe (app) who)
    let read := fun choice : (model).Choice who info =>
      let response := choice.1.getD (⟨none⟩ : (app).Action)
      (resolutionResponseDecision? service.setup service.leaks event response,
        execution.respond (app) who response)
    (service.completedInformationWaitProfile profile weight nonnegative small delta
      deltaNonnegative deltaSmall continuation who info).map read =
      mix delta deltaNonnegative deltaSmall
        (((menu).uniformPolicy (initialLaw service.setup) service.horizon service.scheduler
          who info).map read)
        (mix (weight who info) (nonnegative who info) (small who info)
          (PMF.pure (none, execution.respond (app) who ⟨none⟩))
          ((sourceServiceCanonicalPolicy service.setup service.leaks profile who
            (execution.recall who) (execution.observe (app) who)).map fun response =>
              (resolutionResponseDecision? service.setup service.leaks event response,
                execution.respond (app) who response))) := by
  intro info read
  have canonical := service.sourceCompatibleInfo_immediate_decoded_response profile who
    permitted remaining execution trace compatible event turn unrecorded
  have silentCovered : ∀ response ∈ ((app).silentPolicy (execution.recall who)
      (execution.observe (app) who)).support,
      response ∈ (menu).actions who (execution.recall who) (execution.observe (app) who) := by
    intro response supported
    cases (PMF.mem_support_pure_iff _ _).mp supported
    exact service.bounds.canonicalActions_effective (runtime) service.leaks who _ _
      (service.bounds.silence_canonical (runtime) service.leaks who _ _)
  have silent := (menu).restrictPolicy_map_val (initialLaw service.setup) service.horizon
    service.scheduler who (app).silentPolicy _ _ silentCovered
  unfold completedInformationWaitProfile
  rw [ite_eq_left compatible, mix_map, mix_map]
  congr 1
  congr 1
  · have represented := congrArg (PMF.map (fun response : Option (app).Action =>
      (resolutionResponseDecision? service.setup service.leaks event (response.getD ⟨none⟩),
        execution.respond (app) who (response.getD ⟨none⟩)))) silent
    simpa only [PMF.map_comp, Function.comp_def, ReactiveApplication.silentPolicy,
      PMF.pure_map, Option.getD_some, read, info, resolutionResponseDecision?,
      Option.bind_none] using represented
  · change (service.effectiveImmediateComparator profile who info).map
      ((fun response => (resolutionResponseDecision? service.setup service.leaks event response,
        execution.respond (app) who response)) ∘
          (fun choice => choice.1.getD (⟨none⟩ : (app).Action))) = _
    rw [← PMF.map_comp, canonical]

private theorem bind_fixed_mix {History Result : Type}
    (law : PMF History) (weight : ℝ) (nonnegative : 0 ≤ weight) (small : weight ≤ 1)
    (first second : History → PMF Result) :
    law.bind (fun history => mix weight nonnegative small (first history) (second history)) =
      mix weight nonnegative small (law.bind first) (law.bind second) := by
  let lottery := mix weight nonnegative small (PMF.pure true) (PMF.pure false)
  have exchange := PMF.bind_comm law lottery
    (fun history selected => if selected then first history else second history)
  simpa only [lottery, mix_bind, PMF.pure_bind, Bool.false_eq_true, ↓reduceIte] using exchange

/-- Same-draw initial parameter, effective source prefix, physical resolution
decision and traffic under the actual native assessment. Each hidden prefix's
actual residual identifies separate canonical FALSE and TRUE traffic kernels. -/
def ResolutionConditionalResponseLaw
    {Parameter : Type} (parameter : State L service.setup.context → Parameter)
    (profile : BehavioralProfile service.setup.program) (who : Player) (payload : L.Ty)
    (event : (graph service.setup).EventId)
    (site : (model).InformationSite who)
    (past : List (app).PlayerEntry) (view : (app).PlayerView)
    (weight : Player → (app).Info → ℝ)
    (nonnegative : ∀ player info, 0 ≤ weight player info)
    (small : ∀ player info, weight player info ≤ 1)
    (delta : ℝ) (deltaNonnegative : 0 ≤ delta) (deltaSmall : delta ≤ 1)
    (assessment : (model).BehavioralAssessment) : Prop :=
    ∃ execution : (model).InformationHistory who site.1 → (app).Execution,
      (∀ history, ∃ remaining initial decodedSource,
        history.1.state = some ⟨remaining, some who, execution history⟩ ∧
        (execution history).recall who = past ∧ (execution history).observe (app) who = view ∧
        initial ∈ service.setup.initialLaw.support ∧
        sourceInitialReadout service.setup (execution history).application.config = some initial ∧
        sourceServicePrefix? service.setup event.val (execution history).application.config =
          some decodedSource ∧
        ∃ source : RevealSource service.setup profile event (execution history).application.config,
          source.owner = who ∧ source.payload = payload ∧
          (sourceServiceCanonicalPolicy service.setup service.leaks profile who
            ((execution history).recall who) ((execution history).observe (app) who)).map
              (fun response =>
                (resolutionResponseDecision? service.setup service.leaks event response,
                  (execution history).respond (app) who response)) =
            (revealKernel source.residual (source.source.view source.owner)).map fun disclose =>
              (some disclose, (execution history).respond (app) who
                ((runtime).canonicalServiceDecision service.leaks who
                  ((execution history).recall who)
                  ((execution history).observe (app) who) event
                  (cast (congrArg EventGraph.EventField.Action source.outputEq.symm) disclose)))) ∧
      let belief := assessment.belief who site
      let carry := fun history : (model).InformationHistory who site.1 =>
        ((sourceInitialReadout service.setup (execution history).application.config).map parameter,
          sourceServicePrefix? service.setup event.val (execution history).application.config)
      let read := fun history (choice : (model).Choice who site.1) =>
        let response := choice.1.getD (⟨none⟩ : (app).Action)
        (carry history, (resolutionResponseDecision? service.setup service.leaks event response,
          (runtime).bindingTraffic service.leaks who
            ((execution history).respond (app) who response)))
      let uniform := fun history =>
        ((menu).uniformPolicy (initialLaw service.setup) service.horizon service.scheduler
          who site.1).map (read history)
      let wait := fun history => (carry history, (none,
        (runtime).bindingTraffic service.leaks who
          ((execution history).respond (app) who ⟨none⟩)))
      let draw := fun history =>
        (sourceServiceCanonicalPolicy service.setup service.leaks profile who
          ((execution history).recall who) ((execution history).observe (app) who)).map
          fun response =>
            (carry history, (resolutionResponseDecision? service.setup service.leaks event response,
              (runtime).bindingTraffic service.leaks who
                ((execution history).respond (app) who response)))
      let actual := belief.bind fun history => (assessment.strategy who site.1).map (read history)
      actual = mix delta deltaNonnegative deltaSmall (belief.bind uniform)
        (mix (weight who site.1) (nonnegative who site.1) (small who site.1)
          (belief.map wait) (belief.bind draw)) ∧
      ∀ result, (actual result).toReal =
        ∑ history : (model).InformationHistory who site.1,
          ((model).counterfactualReachProbability assessment.strategy who history.1.trace /
            service.counterfactualInformationMass (menu) assessment.strategy who site Set.univ) *
          (delta * (uniform history result).toReal + (1 - delta) *
            (weight who site.1 * ((PMF.pure (wait history)) result).toReal +
              (1 - weight who site.1) * (draw history result).toReal))

/-- Actual Bayes consistency and the prescribed pin identify this conditional
response law. Source residuals and packet Booleans are derived from full-menu
histories, including retained earlier waits and privately risky foreign paths. -/
theorem information_wait_resolution_bayes_joint_response
    {Parameter : Type} (parameter : State L service.setup.context → Parameter)
    (profile : BehavioralProfile service.setup.program) (who : Player) (payload : L.Ty)
    (permitted : (profile who).Admitted service.setup.program (CommitmentInterface.values _))
    (effective : ∀ player, (profile player).EffectiveDisclosures service.setup.program []
      (Revelations.initial service.setup.context))
    (event : (graph service.setup).EventId)
    (publication : (graph service.setup).outputLayout event = .publication payload)
    (site : (model).InformationSite who)
    (past : List (app).PlayerEntry) (view : (app).PlayerView)
    (observed : site.1 = some (past, view))
    (compatible : service.sourceCompatibleInfo who site.1)
    (turn : view.application.publicView.ownTurn? who = some event)
    (unrecorded : (runtime).eventRecorded service.leaks past event = false)
    (weight : Player → (app).Info → ℝ)
    (nonnegative : ∀ player info, 0 ≤ weight player info)
    (small : ∀ player info, weight player info ≤ 1)
    (delta : ℝ) (deltaNonnegative : 0 ≤ delta) (deltaSmall : delta ≤ 1)
    (assessment : (model).BehavioralAssessment)
    (bayes : InformationModel.BehavioralAssessment.IsBayesConsistent (model) assessment
      ((menu).decisionInformationAntichain (initialLaw service.setup) service.horizon
        service.scheduler))
    (pinned : assessment.strategy who site.1 = service.completedInformationWaitProfile profile
      weight nonnegative small delta deltaNonnegative deltaSmall assessment.strategy who site.1)
    (positive : 0 < (model).informationMass assessment.strategy who site) :
    service.ResolutionConditionalResponseLaw parameter profile who payload event site
      past view weight nonnegative small delta deltaNonnegative deltaSmall assessment := by
  rcases site with ⟨info, decision⟩
  dsimp only at observed
  subst info
  let site : (model).InformationSite who := ⟨some (past, view), decision⟩
  have observed : site.1 = some (past, view) := rfl
  unfold ResolutionConditionalResponseLaw
  let origin := fun history : (model).InformationHistory who site.1 =>
    sourceServiceInformation_owned_prefix service.setup service.leaks (menu)
      (initialLaw service.setup) service.horizon service.scheduler who event past view turn
      ⟨history.1, history.2.trans observed⟩
  let remaining := fun history => Classical.choose (origin history)
  let execution := fun history => Classical.choose (Classical.choose_spec (origin history))
  have actual history := Classical.choose_spec (Classical.choose_spec (origin history))
  have trace history : ((menu).protocol (initialLaw service.setup) service.horizon
      service.scheduler).Trace (some ⟨remaining history, some who, execution history⟩) :=
    (actual history).1 ▸ history.1.trace
  have turnAt history : (execution history).application.publicView.ownTurn? who = some event := by
    change ((execution history).observe (app) who).application.publicView.ownTurn? who = _
    rw [(actual history).2.2.1]
    exact turn
  have inputAt history : some ((execution history).recall who,
      (execution history).observe (app) who) = site.1 := by
    rw [(actual history).2.1, (actual history).2.2.1, observed]
  refine ⟨execution, ?_, ?_⟩
  · intro history
    have rawTrace := (menu).toRawTrace (initialLaw service.setup) service.horizon
      service.scheduler (trace history)
    obtain ⟨initial, supported, initialized⟩ := sourceInitialReadout_history service.setup
      service.leaks service.horizon service.scheduler
        ⟨remaining history, some who, execution history⟩ rawTrace
    have ready := ((execution history).application.publicView_eventReady event).mp
      (PublicView.ownTurn?_spec _ who event (turnAt history)).1
    obtain ⟨residual⟩ := menu_ready_sourceResidual service.setup service.leaks profile (menu)
      service.horizon service.scheduler (trace history)
        ⟨remaining history, some who, execution history⟩ rfl event ready
    obtain ⟨source⟩ := residual.revealSource publication
    change RevealSource service.setup profile event (execution history).application.config at source
    have ownerEq : source.owner = who := Option.some.inj (source.owned.symm.trans
      (PublicView.ownTurn?_spec _ who event (turnAt history)).2)
    have outputEq : source.payload = payload :=
      EventGraph.EventField.publication.inj (source.outputEq.symm.trans publication)
    have kernel := service.canonical_resolution_joint profile effective event (execution history)
      source (remaining history) (ownerEq.symm ▸ trace history) ready
    refine ⟨remaining history, initial,
      residual.lift (ProtocolState.entry residual.program residual.source),
      (actual history).1, (actual history).2.1, (actual history).2.2.1,
      supported, initialized, residual.decode, source, ownerEq, outputEq, ?_⟩
    simpa only [ownerEq] using kernel
  · intro belief carry read uniform wait draw actualLaw
    have conditional history :
        (assessment.strategy who site.1).map (read history) =
          mix delta deltaNonnegative deltaSmall (uniform history)
            (mix (weight who site.1) (nonnegative who site.1) (small who site.1)
              (PMF.pure (wait history)) (draw history)) := by
      have rawLaw := service.resolution_pin_joint profile who permitted event (execution history)
        (remaining history) (trace history) ((inputAt history).symm ▸ compatible) (turnAt history)
        (by rw [(actual history).2.1]; exact unrecorded) weight nonnegative small delta
        deltaNonnegative deltaSmall assessment.strategy
      dsimp only at rawLaw
      let physicalRead := fun response : (app).Action =>
        (carry history, (resolutionResponseDecision? service.setup service.leaks event response,
          (runtime).bindingTraffic service.leaks who
            ((execution history).respond (app) who response)))
      have mapped := congrArg (PMF.map (fun pair =>
        (carry history, (pair.1, (runtime).bindingTraffic service.leaks who pair.2)))) rawLaw
      simp only [mix_map, PMF.map_comp, PMF.pure_map, Function.comp_def] at mapped
      have pinInput := congrArg (fun info =>
        (service.completedInformationWaitProfile profile weight nonnegative small delta
          deltaNonnegative deltaSmall assessment.strategy who info).map
            (fun choice => physicalRead (choice.1.getD (⟨none⟩ : (app).Action))))
              (inputAt history)
      have uniformInput := congrArg (fun info =>
        ((menu).uniformPolicy (initialLaw service.setup) service.horizon service.scheduler
          who info).map
            (fun choice => physicalRead (choice.1.getD (⟨none⟩ : (app).Action))))
              (inputAt history)
      rw [pinInput, uniformInput, inputAt history] at mapped
      rw [pinned]
      exact mapped
    have equal := (InformationModel.BehavioralAssessment.isBayesConsistentAt_iff (model)
      assessment who site ((menu).decisionInformationAntichain (initialLaw service.setup)
        service.horizon service.scheduler who site) positive).mp (bayes who site positive)
    constructor
    · calc
        actualLaw = belief.bind (fun history => mix delta deltaNonnegative deltaSmall
            (uniform history) (mix (weight who site.1) (nonnegative who site.1) (small who site.1)
              (PMF.pure (wait history)) (draw history))) := by
                apply bind_congr_on_support
                intro history _
                exact conditional history
        _ = _ := by
          rw [bind_fixed_mix, bind_fixed_mix]
          exact congrArg (fun waiting => mix delta deltaNonnegative deltaSmall (belief.bind uniform)
            (mix (weight who site.1) (nonnegative who site.1) (small who site.1)
              waiting (belief.bind draw))) (PMF.bind_pure_comp wait belief)
    · intro result
      rw [toReal_bind_apply, expect_eq_sum]
      apply Finset.sum_congr rfl
      intro history _
      rw [conditional history, mix_apply_toReal, mix_apply_toReal]
      change (assessment.belief who site history).toReal * _ = _
      rw [equal, service.bayesBelief_apply_counterfactual (menu) assessment.strategy who site
        positive history]
